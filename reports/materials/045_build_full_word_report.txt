normal_style.element.rPr.rFonts.set(qn('w:eastAsia'), 'Times New Roman')

def set_run_font(run, size=13, bold=False, italic=False):
    run.font.name = 'Times New Roman'
    run.font.size = Pt(size)
    run.font.color.rgb = RGBColor(0, 0, 0)
    run.bold = bold
    run.italic = italic
    run.element.rPr.rFonts.set(qn('w:eastAsia'), 'Times New Roman')

def add_p(text="", bold=False, italic=False, size=13, align=WD_ALIGN_PARAGRAPH.LEFT, space_after=4, space_before=0):
    p = doc.add_paragraph()
    p.alignment = align
    p.paragraph_format.space_after = Pt(space_after)
    p.paragraph_format.space_before = Pt(space_before)
    p.paragraph_format.line_spacing = 1.15
    if text:
        run = p.add_run(text)
        set_run_font(run, size=size, bold=bold, italic=italic)
    return p

def add_heading_1(text):
    return add_p(text, bold=True, size=15, space_before=12, space_after=6)

def add_heading_2(text):
    return add_p(text, bold=True, size=13.5, space_before=8, space_after=4)

def add_image_cell(img_name, width_in=6.3):
    img_path = os.path.join(img_dir, img_name)
    if os.path.exists(img_path):
        p = doc.add_paragraph()
        p.alignment = WD_ALIGN_PARAGRAPH.CENTER
        p.paragraph_format.space_after = Pt(6)
        p.paragraph_format.space_before = Pt(4)
        run = p.add_run()
        run.add_picture(img_path, width=Inches(width_in))
    else:
        print(f"Warning: image {img_name} not found!")
