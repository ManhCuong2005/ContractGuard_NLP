# Thư viện chuẩn hóa Unicode tiếng Việt
import unicodedata

# 1. Hàm chuẩn hóa văn bản cơ bản
def normalize_basic(text):

    # Nếu dữ liệu thiếu thì trả về chuỗi rỗng
    if text is None or pd.isna(text):
        return ""

    # Chuyển dữ liệu thành chuỗi
    text = str(text)

    # Chuẩn hóa Unicode về NFC
    text = unicodedata.normalize("NFC", text)

    # Thay khoảng trắng đặc biệt bằng khoảng trắng thường
    text = text.replace("\u00A0", " ")

    # Gộp các khoảng trắng liên tiếp
    text = re.sub(r"\s+", " ", text)

    # Trả về chuỗi đã xóa khoảng trắng hai đầu
    return text.strip()


# 2. Hàm xóa dấu câu
def remove_punctuation(text):

    # Thay các ký tự không phải chữ, số hoặc khoảng trắng
    # bằng một dấu cách
    text = re.sub(r"[^\w\s]", " ", text, flags=re.UNICODE)

    # Gộp khoảng trắng sau khi xóa dấu câu
    return re.sub(r"\s+", " ", text).strip()


# 3. Hàm xóa số
def remove_numbers(text):

    # Xóa các cụm bắt đầu bằng số
    text = re.sub(r"\b\d[\d.,]*\b", " ", text)

    # Gộp khoảng trắng
    return re.sub(r"\s+", " ", text).strip()


# 4. Hàm tiền xử lý tổng hợp
def preprocess_text(text):

    # Chuẩn hóa văn bản
    text = normalize_basic(text)

    # Chuyển toàn bộ thành chữ thường
    text = text.lower()

    # Xóa các số
    text = remove_numbers(text)

    # Xóa dấu câu
    text = remove_punctuation(text)

    # Trả kết quả
    return text


# 5. Áp dụng tiền xử lý lên cột text
