const work = 'D:/NLP/Lab09/report_work';
const esc = s => String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
const nb = JSON.parse(fs.readFileSync(path.join(work,'Lab09_executed.ipynb'),'utf8'));
const source = fs.readFileSync(path.join(work,'ridebot.py'),'utf8').replace(/\r\n/g,'\n');
const txt = s => Array.isArray(s) ? s.join('') : s;
const blocks = [];
function add(id,label,html){ blocks.push({id,label,html}); }
function code(id,label,s){ add(id,label,`<pre>${esc(s)}</pre>`); }
for(const cell of nb.cells.filter(c=>c.cell_type==='code' && c.metadata.original_index<28)){
  const i = cell.metadata.original_index;
  code('c'+i,`Cell ${i+1} · Code`,txt(cell.source));
  let n=0;
  for(const out of cell.outputs || []) {
    if(out.name==='stderr' && txt(out.text).includes('building the font cache')) continue;
    const id = `o${i}_${n++}`;
    if(out.output_type==='stream') {
      const text=txt(out.text);
      if(i===17 && text.includes('User:')) {
        text.trim().split(/\n\s*\n/).filter(s=>s.trim()).forEach((s,j)=>code(id+'_'+j,`Cell ${i+1} · Lượt ${j+1}`,s.trim()));
      } else if(i===20 && text.includes('CASE')) {
        text.split(/(?=\nCASE \d)/).filter(s=>s.trim()).forEach((s,j)=>code(id+'_'+j,`Cell ${i+1} · Kết quả ${j+1}`,s.trim()));
      } else code(id,`Cell ${i+1} · Kết quả`,text.trim());
    } else if(out.data) {
      if(out.data['image/png']) add(id,`Cell ${i+1} · Kết quả`,`<img class="plot" src="data:image/png;base64,${txt(out.data['image/png'])}">`);
      else if(out.data['text/html']) add(id,`Cell ${i+1} · Kết quả`,txt(out.data['text/html']));
      else if(out.data['text/plain']) code(id,`Cell ${i+1} · Kết quả`,txt(out.data['text/plain']));
    }
  }
}
