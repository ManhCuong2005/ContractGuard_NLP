def get_nonzero_features(
    X,
    feature_names,
    row_idx,
    top_n=20
):
    row = X.getrow(row_idx)

    pairs = [
        (feature_names[i], value)
        for i, value in zip(row.indices, row.data)
    ]

    pairs = sorted(
        pairs,
        key=lambda x: x[1],
        reverse=True
    )

    return pd.DataFrame(
        pairs[:top_n],
        columns=["feature", "value"]
    )

def top_tfidf_features(
    X,
    feature_names,
    row_idx,
    top_n=10
):
    row = X.getrow(row_idx)

    pairs = [
        (feature_names[i], float(value))
        for i, value in zip(row.indices, row.data)
    ]

    pairs = sorted(
        pairs,
        key=lambda x: x[1],
        reverse=True
    )

    return pd.DataFrame(
        pairs[:top_n],
        columns=["feature", "tfidf"]
    )

def find_similar_articles(
    doc_index,
    top_k,
    similarity_matrix,
    data
):
    scores = similarity_matrix[doc_index]

    ranked = np.argsort(scores)[::-1]

    ranked = [
        idx for idx in ranked
        if idx != doc_index
    ]

    ranked = ranked[:top_k]

    columns = ["title", "processed_text"]
    if "category" in data.columns:
        columns.insert(1, "category")

    result = data.iloc[ranked][columns].copy()

    result["similarity"] = [
        float(scores[idx])
        for idx in ranked
    ]

    return result
