# LAB 01 — TÁCH CÂU / TÁCH TỪ / TỪ LOẠI / THỰC THỂ / TRÍCH THEO MẪU

# ──────────────────────────────────────────────────────────────────────────────
# 01.02a  ·  Cài thư viện còn thiếu
# ──────────────────────────────────────────────────────────────────────────────
# find_spec() kiểm tra thư viện; check_call() gọi pip để cài nếu chưa có.
import importlib.util, subprocess, sys


# ──────────────────────────────────────────────────────────────────────────────
# 01.02b  ·  Cài thư viện còn thiếu — áp dụng
# ──────────────────────────────────────────────────────────────────────────────
# find_spec() kiểm tra thư viện; check_call() gọi pip để cài nếu chưa có.
def ensure_package(name):
    if importlib.util.find_spec(name) is None:
        subprocess.check_call([sys.executable, "-m", "pip", "install", "-q", name])


# ──────────────────────────────────────────────────────────────────────────────
# 01.02c  ·  Cài thư viện còn thiếu — áp dụng
# ──────────────────────────────────────────────────────────────────────────────
# find_spec() kiểm tra thư viện; check_call() gọi pip để cài nếu chưa có.
ensure_package("underthesea")
ensure_package("pandas")


# ──────────────────────────────────────────────────────────────────────────────
# 01.03  ·  Thư viện NLP tiếng Việt
# ──────────────────────────────────────────────────────────────────────────────
# underthesea cung cấp tách câu, tách từ, từ loại và nhận diện thực thể; pandas tạo bảng.
from IPython.display import display
import re
import pandas as pd
from underthesea import sent_tokenize, word_tokenize, pos_tag, ner
print("Import thành công.")


# ──────────────────────────────────────────────────────────────────────────────
# 01.04  ·  Văn bản đầu vào
# ──────────────────────────────────────────────────────────────────────────────
# Chuỗi ba nháy chứa văn bản nhiều dòng; strip() bỏ khoảng trắng ở hai đầu.
text = """
Ngày 12/8/2026, Công ty ABC khai trương văn phòng mới tại Đà Nẵng.
Giám đốc Nguyễn Văn An cho biết công ty dự kiến tuyển thêm 100 kỹ sư phần mềm trong năm nay.
Doanh nghiệp sẽ hợp tác với Trường Đại học Công nghệ Thông tin và Truyền thông Việt - Hàn.
Mai công ty sẽ bắt đầu nhận hồ sơ.
Ứng viên có thể liên hệ qua email tuyendung@example.com hoặc số điện thoại 0905123456.
""".strip()
print(text)


# ──────────────────────────────────────────────────────────────────────────────
# 01.05  ·  Tách câu
# ──────────────────────────────────────────────────────────────────────────────
# sent_tokenize() chia một đoạn văn thành các câu riêng.
# Ví dụ: "Tôi học. Bạn đọc." → ["Tôi học.", "Bạn đọc."]
sentences = sent_tokenize(text)
print("Số câu:", len(sentences))
for i, s in enumerate(sentences, 1):
    print(f"{i}. {s}")


# ──────────────────────────────────────────────────────────────────────────────
# 01.06  ·  Tách từ tiếng Việt
# ──────────────────────────────────────────────────────────────────────────────
# word_tokenize() nhận diện từ tiếng Việt; split() chỉ cắt theo khoảng trắng. format="text" trả
# chuỗi có dấu _ trong từ ghép.
# Ví dụ: "sinh viên học" → "sinh_viên học" (minh họa).
sample_sentence = "Giám đốc Nguyễn Văn An cho biết công ty dự kiến tuyển thêm 100 kỹ sư phần mềm trong năm nay."
print("split():")
print(sample_sentence.split())
print("\nword_tokenize():")
print(word_tokenize(sample_sentence))
print("\nformat='text':")
print(word_tokenize(sample_sentence, format="text"))


# ──────────────────────────────────────────────────────────────────────────────
# 01.07  ·  Xác định từ loại
# ──────────────────────────────────────────────────────────────────────────────
# pos_tag() gán từ loại cho từng từ; DataFrame() trình bày kết quả thành bảng.
# Ví dụ: N: danh từ; V: động từ; A: tính từ; Np: tên riêng.
pos_result = pos_tag(sample_sentence)
pd.DataFrame(pos_result, columns=["Token", "POS"])


# ──────────────────────────────────────────────────────────────────────────────
# 01.08  ·  Nhận diện thực thể
# ──────────────────────────────────────────────────────────────────────────────
# ner() tìm các tên người, địa điểm, tổ chức trong văn bản.
# Ví dụ: "Lan ở Hà Nội" → Lan: người; Hà Nội: địa điểm (minh họa).
ner_result = ner(text)
for item in ner_result[:30]:
    print(item)


# ──────────────────────────────────────────────────────────────────────────────
# 01.09  ·  Đọc kết quả NER
# ──────────────────────────────────────────────────────────────────────────────
# DataFrame() tổ chức kết quả NER thành các cột để xem token và nhãn.
# Ví dụ: B-PER: bắt đầu tên người; I-PER: tiếp tục tên người; O: ngoài thực thể.
if ner_result and len(ner_result[0]) == 4:
    ner_df = pd.DataFrame(ner_result, columns=["Token", "POS", "Chunk", "NER"])
else:
    ner_df = pd.DataFrame(ner_result)
ner_df.head(60)


# ──────────────────────────────────────────────────────────────────────────────
# 01.10  ·  Trích email và số điện thoại
# ──────────────────────────────────────────────────────────────────────────────
# re.findall() lấy tất cả phần văn bản khớp mẫu Regex.
# Ví dụ: Mẫu \d+ trích các cụm chữ số, ví dụ "có 12 sách" → ["12"].
email_pattern = r"\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b"
phone_pattern = r"\b0\d{9}\b"
print("Email:", re.findall(email_pattern, text))
print("Phone:", re.findall(phone_pattern, text))


# ──────────────────────────────────────────────────────────────────────────────
# 01.11  ·  Thử NER theo ngữ cảnh
# ──────────────────────────────────────────────────────────────────────────────
# ner() được áp dụng trên các câu mơ hồ để xem cùng một chữ được hiểu khác nhau ra sao.
# Ví dụ: "Mai" có thể là tên người hoặc chỉ ngày mai.
challenge_sentences = [
    "Mai tôi đi Đà Nẵng.",
    "Mai đang học NLP tại trường.",
    "Hôm nay Bình đến Hòa Bình.",
    "Công ty An Bình mở chi nhánh mới tại Huế."
]
for s in challenge_sentences:
    print("\n" + "="*70)
    print("TEXT:", s)
    for item in ner(s):
        print(item)


# ──────────────────────────────────────────────────────────────────────────────
# 01.12  ·  Văn bản thực hành
# ──────────────────────────────────────────────────────────────────────────────
# Đoạn văn nhiều câu dùng làm input cho các thao tác NLP bên dưới; strip() bỏ khoảng trắng hai
# đầu.
student_text = """
Sáng ngày 15/8/2026, Trường Đại học Bách khoa - Đại học Đà Nẵng tổ chức Ngày hội tuyển dụng công nghệ tại quận Hải Châu.
Sự kiện thu hút hơn 30 doanh nghiệp phần mềm và khoảng 800 sinh viên tham dự.
Bà Nguyễn Thu Hà, đại diện Công ty FPT Software, giới thiệu hơn 120 vị trí thực tập và việc làm toàn thời gian.
Nhiều sinh viên quan tâm đến các công việc về trí tuệ nhân tạo, phân tích dữ liệu và an ninh mạng.
Bạn Minh cho biết mình muốn tìm một môi trường làm việc có cơ hội học hỏi lâu dài.
Trong khi đó, Mai dự định nộp hồ sơ vào vị trí kỹ sư kiểm thử phần mềm.
Các nhà tuyển dụng đánh giá cao kỹ năng giao tiếp, tư duy giải quyết vấn đề và khả năng làm việc nhóm.
Đại diện Microsoft Việt Nam cho rằng sinh viên nên xây dựng hồ sơ dự án cá nhân từ năm thứ hai.
Ban tổ chức cũng mở một phiên tư vấn CV miễn phí tại hội trường A5.
Người tham dự có thể đăng ký phỏng vấn nhanh qua email career@bkdn.edu.vn hoặc số điện thoại 0905123456.
Chương trình kết thúc lúc 16 giờ và sẽ được tổ chức lại vào tháng 11.

print(student_text)


# ──────────────────────────────────────────────────────────────────────────────
# 01.13  ·  Tách câu trên văn bản mới
# ──────────────────────────────────────────────────────────────────────────────
# sent_tokenize() dùng được với bất kỳ đoạn văn mới nào để tạo danh sách câu.
student_sentences = sent_tokenize(student_text)
print("Số câu:", len(student_sentences))
for i, s in enumerate(student_sentences, 1):
    print(f"{i}. {s}")


# ──────────────────────────────────────────────────────────────────────────────
# 01.14  ·  Tách từ cho nhiều câu
# ──────────────────────────────────────────────────────────────────────────────
# word_tokenize(..., format="text") tạo chuỗi từ đã ghép âm tiết cho từng câu.
# Ví dụ: "ngôn ngữ tự nhiên" có thể thành "ngôn_ngữ_tự_nhiên".
for s in student_sentences[:3]:
    print("\nTEXT:", s)
    print("TOKENS:", word_tokenize(s, format="text"))


# ──────────────────────────────────────────────────────────────────────────────
# 01.15  ·  Xem từ loại của một câu
# ──────────────────────────────────────────────────────────────────────────────
# pos_tag() xác định danh từ, động từ, tên riêng... trong câu; DataFrame() tạo bảng dễ đối
# chiếu.
if student_sentences:
    selected_sentence = student_sentences[2] if len(student_sentences) >= 3 else student_sentences[0]
    display(pd.DataFrame(pos_tag(selected_sentence), columns=["Token", "POS"]))
else:
    print("Chưa có dữ liệu hợp lệ.")


# ──────────────────────────────────────────────────────────────────────────────
# 01.16  ·  Tìm thực thể trong văn bản mới

student_ner = ner(student_text)
if student_ner and len(student_ner[0]) == 4:
    student_ner_df = pd.DataFrame(student_ner, columns=["Token", "POS", "Chunk", "NER"])
else:
    student_ner_df = pd.DataFrame(student_ner)
student_ner_df.head(80)


# ──────────────────────────────────────────────────────────────────────────────
# 01.17  ·  Kiểm tra trường hợp NER mơ hồ
# ──────────────────────────────────────────────────────────────────────────────
# ner() cho dự đoán theo ngữ cảnh; DataFrame() đặt kết quả cạnh cách hiểu của người đọc để nhận
# ra lỗi.
review_cases = [
    ("Mai tôi đi Đà Nẵng.", "Mai", "NER", "Trong câu này, Mai có thể chỉ ngày mai; cần đọc ngữ cảnh thay vì mặc định là người."),
    ("Hôm nay Bình đến Hòa Bình.", "Bình / Hòa Bình", "NER", "Bình là tên người, Hòa Bình là địa điểm; cần phân biệt cùng âm tiết ở hai vị trí."),
    ("Công ty An Bình mở chi nhánh mới tại Huế.", "Công ty An Bình", "NER", "An Bình thuộc tên tổ chức trong ngữ cảnh này; không tự tách thành tên người."),
]
analysis_rows = []
for sentence, target, task, judgment in review_cases:

    analysis_rows.append({

        "Task": task,
        "Model output": actual_output,
        "Expected / Human judgment": judgment,
        "Đúng/Sai/Tranh luận": "Đối chiếu output thực tế với ngữ cảnh; có thể tranh luận khi thiếu ngữ cảnh.",

    })
student_error_analysis = pd.DataFrame(analysis_rows)
display(student_error_analysis)

# ──────────────────────────────────────────────────────────────────────────────
# 02.02b  ·  Cài thư viện — áp dụng
# ──────────────────────────────────────────────────────────────────────────────
# find_spec() kiểm tra gói; check_call() cài requests, BeautifulSoup và pandas nếu còn thiếu.
def ensure_package(package_name, pip_name=None):
    pip_name = pip_name or package_name
    if importlib.util.find_spec(package_name) is None:
        subprocess.check_call([
            sys.executable, "-m", "pip", "install", "-q", pip_name
        ])


# ──────────────────────────────────────────────────────────────────────────────
# 02.02c  ·  Cài thư viện — áp dụng
# ──────────────────────────────────────────────────────────────────────────────
# find_spec() kiểm tra gói; check_call() cài requests, BeautifulSoup và pandas nếu còn thiếu.
ensure_package("requests")
ensure_package("bs4", "beautifulsoup4")

# ──────────────────────────────────────────────────────────────────────────────
# 02.03  ·  Thư viện thu thập dữ liệu
# ──────────────────────────────────────────────────────────────────────────────
# requests tải web; BeautifulSoup đọc HTML; re xử lý mẫu chuỗi; urljoin ghép URL.
import re

# ──────────────────────────────────────────────────────────────────────────────
# 02.04  ·  HTML mẫu để thực hành
# ──────────────────────────────────────────────────────────────────────────────
# Chuỗi HTML mô phỏng trang bài viết và chuyên mục, giúp thử trích nội dung mà không cần mạng.
SAMPLE_ARTICLE_HTML = """
<!DOCTYPE html>
<html lang="vi">
<head>
    <meta charset="utf-8">
    <title>Demo News</title>
</head>
<body>
    <header>
        <nav>Trang chủ | Kinh doanh | Công nghệ | Thể thao</nav>
    </header>

    <main>
        <article class="article-detail">
            <div class="category">Kinh doanh</div>
            <h1 class="title">Doanh nghiệp ABC mở trung tâm công nghệ mới tại Đà Nẵng</h1>

            <p class="description">
                Trung tâm mới dự kiến tuyển thêm 300 kỹ sư phần mềm trong hai năm tới.
            </p>

            <time class="published">2026-08-20 08:30</time>

            <div class="article-body">
                <p>Công ty ABC vừa khai trương trung tâm công nghệ mới tại Đà Nẵng.</p>
                <p>Doanh nghiệp cho biết dự án tập trung vào trí tuệ nhân tạo và xử lý dữ liệu.</p>
                <p>Trong giai đoạn đầu, công ty dự kiến tuyển khoảng 300 kỹ sư phần mềm.</p>
            </div>
        </article>
    </main>

    <footer>© Demo News</footer>
</body>
</html>
"""
SAMPLE_CATEGORY_HTML = """
<!DOCTYPE html>
<html lang="vi">
<body>
    <section class="article-list">
        <a class="article-link" href="/kinh-doanh/doanh-nghiep-abc.html">Bài 1</a>
        <a class="article-link" href="/cong-nghe/ai-viet-nam.html">Bài 2</a>
        <a class="article-link" href="/the-thao/doi-tuyen.html">Bài 3</a>
        <a class="article-link" href="/kinh-doanh/doanh-nghiep-abc.html">Bài 1 bị lặp</a>
        <a href="/about-us">Giới thiệu</a>
    </section>
</body>
</html>
"""
print("Đã tạo HTML mẫu dùng offline.")


# ──────────────────────────────────────────────────────────────────────────────
# 02.05  ·  Đọc cấu trúc HTML
# ──────────────────────────────────────────────────────────────────────────────
# BeautifulSoup() biến HTML thành cây thẻ; get_text() lấy phần chữ trong thẻ.
# Ví dụ: <title>Tin mới</title> → "Tin mới".
soup = BeautifulSoup(SAMPLE_ARTICLE_HTML, "html.parser")
print(soup.title.get_text(strip=True))


# ──────────────────────────────────────────────────────────────────────────────
# 02.06  ·  Trích nội dung bài viết
# ──────────────────────────────────────────────────────────────────────────────
# select_one() tìm một thẻ, select() tìm nhiều thẻ, get_text() lấy chữ; join() ghép các đoạn.
# Ví dụ: ".title" chọn class title; ".article-body p" chọn các đoạn trong thân bài.
category = soup.select_one(".category").get_text(" ", strip=True)
title = soup.select_one(".title").get_text(" ", strip=True)
description = soup.select_one(".description").get_text(" ", strip=True)
published_at = soup.select_one(".published").get_text(" ", strip=True)
paragraphs = [
    p.get_text(" ", strip=True)
    for p in soup.select(".article-body p")
]
text = " ".join(paragraphs)
print("CATEGORY:", category)
print("TITLE:", title)
print("DESCRIPTION:", description)
print("PUBLISHED:", published_at)
print("TEXT:", text)


# ──────────────────────────────────────────────────────────────────────────────
# 02.07a  ·  Làm gọn khoảng trắng — hàm clean_extracted_text()
# ──────────────────────────────────────────────────────────────────────────────
# Làm gọn khoảng trắng trong văn bản trích từ HTML.
# Ví dụ: '  xin   chào\n ' → 'xin chào'; None → ''.
def clean_extracted_text(text):
    if text is None:
        return ""
    return re.sub(r"\s+", " ", str(text)).strip()


# ──────────────────────────────────────────────────────────────────────────────
# 02.07b  ·  Làm gọn khoảng trắng — áp dụng
# ──────────────────────────────────────────────────────────────────────────────

# 02.10a  ·  Lọc link bài viết — hàm is_article_url()
# ──────────────────────────────────────────────────────────────────────────────
# Nhận biết một link có đúng dạng URL bài viết cần lấy hay không.
# Ví dụ: Có thể chỉ chấp nhận link kết thúc .html và nằm trong các chuyên mục cho phép; trả
# True/False.
def is_article_url(url):
    if not url:
        return False
    allowed_prefixes = (
        "/kinh-doanh/",
        "/cong-nghe/",
        "/the-thao/",
    )
    return url.endswith(".html") and url.startswith(allowed_prefixes)

DEFAULT_HEADERS = {

# 02.13b  ·  Tải trang có thử lại — hàm safe_get()
# ──────────────────────────────────────────────────────────────────────────────
# Tải một trang web với timeout và thử lại khi gặp lỗi mạng/HTTP.
# Ví dụ: Dùng thay requests.get đơn giản khi muốn tự thử lại lúc một lần tải tạm thời thất bại.
def safe_get(url, timeout=20, retries=3):
    last_error = None
    for attempt in range(1, retries + 1):
        try:
            response = requests.get(
                url, headers=DEFAULT_HEADERS, timeout=timeout
            )
            response.raise_for_status()
            response.encoding = response.apparent_encoding or "utf-8"
            return response
        except requests.RequestException as exc:
            last_error = exc
            if attempt < retries:
                time.sleep(attempt * 2)
    raise last_error

# 02.15a  ·  Thu link từ nhiều chuyên mục — hàm collect_article_urls()
# ──────────────────────────────────────────────────────────────────────────────
# Thu danh sách link bài viết từ các trang chuyên mục của một website.
# Ví dụ: Trả danh sách URL bài để tải tiếp và các lỗi gặp khi thu link.
def collect_article_urls(source, pages_per_category=5, delay_seconds=0.5):
    urls = []
    errors = []
    for slug in source["category_slugs"]:
        for page in range(1, pages_per_category + 1):
            category_url = source["page_url"](slug, page)
            try:
                response = safe_get(category_url)
                category_soup = BeautifulSoup(response.text, "html.parser")
                for a in category_soup.select("a[href]"):
                    href = a.get("href", "").strip()
                    absolute_url = urljoin(source["base_url"], href)
                    absolute_url = absolute_url.split("#")[0].split("?")[0]
                    if re.match(source["article_url_regex"], absolute_url):
                        urls.append(absolute_url)
                print(source["name"], slug, page, "->", len(set(urls)), "URLs")
            except Exception as exc:
                errors.append({
                    "url": category_url,
                    "source": source["name"],
                    "stage": "collect_urls",
                    "error": type(exc).__name__ + ": " + str(exc),
                })
            if delay_seconds > 0:
                time.sleep(delay_seconds)
    return list(dict.fromkeys(urls)), errors

# 02.17a  ·  Chuyển HTML thành bài viết — hàm extract_text_or_empty()
# ──────────────────────────────────────────────────────────────────────────────
# Trích chữ từ vùng HTML được chọn; thiếu vùng đó thì trả chuỗi rỗng.
# Ví dụ: Có h1.title → lấy tiêu đề; không có → ''.
def extract_text_or_empty(soup, selector):
    if not selector:
        return ""
    node = soup.select_one(selector)
    return clean_extracted_text(
        node.get_text(" ", strip=True)
    ) if node else ""


# ──────────────────────────────────────────────────────────────────────────────
# 02.17b  ·  Chuyển HTML thành bài viết — hàm parse_article_html()
# ──────────────────────────────────────────────────────────────────────────────
# Trích các trường có cấu trúc của một bài từ HTML.
# Ví dụ: HTML của một bài → dictionary có title, text, category, description... để tạo corpus.
def parse_article_html(
    html,
    url="",
    source="",
    selectors=None,
):
    selectors = selectors or {}
    soup = BeautifulSoup(html, "html.parser")
    category = extract_text_or_empty(
        soup,
        selectors.get("category", "")
    )
    title = extract_text_or_empty(
        soup,
        selectors.get("title", "")
    )
    description = extract_text_or_empty(
        soup,
        selectors.get("description", "")
    )
    published_at = extract_text_or_empty(
        soup,
        selectors.get("published_at", "")
    )
    paragraph_selector = selectors.get("paragraphs", "")

    # select(): Tìm tất cả thẻ khớp CSS selector.
    paragraphs = [
        clean_extracted_text(p.get_text(" ", strip=True))
        for p in soup.select(paragraph_selector)
    ] if paragraph_selector else []
    paragraphs = [p for p in paragraphs if p]
    text = clean_extracted_text(" ".join(paragraphs))
    return {
        "url": url,
        "source": source,
        "category": category,
        "title": title,
        "description": description,
        "text": text,
        "published_at": published_at,
        "crawled_at": datetime.now().isoformat(timespec="seconds"),
    }

# 02.19  ·  Kiểm tra bài hợp lệ
# ──────────────────────────────────────────────────────────────────────────────
# validate_article() phát hiện thiếu URL, thiếu tiêu đề, text rỗng hoặc quá ngắn để dùng làm
# corpus.
def validate_article(record, min_text_length=300):
    problems = []
    if not record.get("url"):
        problems.append("missing_url")
    if not record.get("title"):
        problems.append("missing_title")
    text = record.get("text", "")
    if not text:
        problems.append("empty_text")
    elif len(text) < min_text_length:
        problems.append("text_too_short")
    return {
        "is_valid": len(problems) == 0,
        "problems": problems,
    }


# ──────────────────────────────────────────────────────────────────────────────
# 02.20a  ·  Thu thập nhiều bài — hàm crawl_article()
# ──────────────────────────────────────────────────────────────────────────────
# Tải và trích nội dung của một URL thành một record bài viết.
# Ví dụ: Dùng khi đã có link bài và muốn lấy text/metadata của bài đó.
def crawl_article(url, source, selectors):
    response = safe_get(url)
    return parse_article_html(
        response.text,
        url=url,
        source=source,
        selectors=selectors,
    )


# ──────────────────────────────────────────────────────────────────────────────
# 02.20b  ·  Thu thập nhiều bài — hàm crawl_many()
# ──────────────────────────────────────────────────────────────────────────────
# Thu nội dung của nhiều URL, tách bài hợp lệ khỏi các trường hợp lỗi.
# Ví dụ: Danh sách link → danh sách record bài viết và danh sách lỗi tải/trích/kiểm tra.
def crawl_many(
    urls,
    source,
    selectors,
    delay_seconds=1.0,
    min_text_length=300,
):
    articles = []
    errors = []
    for i, url in enumerate(urls, start=1):
        print(f"[{source} {i}/{len(urls)}] {url}")
        try:
            record = crawl_article(url, source, selectors)
            check = validate_article(record, min_text_length)
            if check["is_valid"]:
                articles.append(record)
            else:
                errors.append({
                    "url": url,
                    "source": source,
                    "stage": "validation",
                    "error": ",".join(check["problems"]),
                })
        except Exception as exc:
            errors.append({
                "url": url,
                "source": source,
                "stage": "crawl_article",
                "error": type(exc).__name__ + ": " + str(exc),
            })
        if delay_seconds > 0:
            time.sleep(delay_seconds)
    return articles, errors

# 02.24  ·  Làm sạch bảng dữ liệu
# ──────────────────────────────────────────────────────────────────────────────
# fillna() thay ô thiếu; drop_duplicates() bỏ bài trùng; str.len() giúp lọc bài quá ngắn.
def clean_dataframe(df, min_text_length=300):
    result = df.copy()
    for col in ["category", "title", "description", "text"]:
        if col in result.columns:
            result[col] = (
                result[col]
                .fillna("")
                .map(clean_extracted_text)
            )
    result = result[
        result["text"].str.len() >= min_text_length
    ]
    result = result.drop_duplicates(
        subset=["url"],
        keep="first"
    )
    result = result.drop_duplicates(
        subset=["text"],
        keep="first"
    )
    return result.reset_index(drop=True)


# ──────────────────────────────────────────────────────────────────────────────
# 02.25  ·  So sánh trước và sau làm sạch
# ──────────────────────────────────────────────────────────────────────────────
# clean_dataframe() áp dụng chuẩn hóa và lọc trùng; value_counts() cho biết số bài mỗi nguồn còn
# lại.
df_clean = clean_dataframe(df, min_text_length=300)
print("Before:", len(df))

df_clean.head()

def ensure_package(package_name, pip_name=None):
    pip_name = pip_name or package_name
    if importlib.util.find_spec(package_name) is None:
        print(f"Đang cài {pip_name} ...")
        subprocess.check_call([
            sys.executable, "-m", "pip", "install", pip_name
        ])
    else:
        print(f"{package_name}: READY")

# ──────────────────────────────────────────────────────────────────────────────
# 03.04a  ·  Quan sát mã Unicode — hàm show_codepoints()
# ──────────────────────────────────────────────────────────────────────────────
# Xem mã Unicode thật của từng ký tự để nhận biết khác biệt chữ/dấu ẩn.
# Ví dụ: Chữ 'ò' dạng ghép và 'o' + dấu huyền rời nhìn giống nhau nhưng có mã ký tự khác.
def show_codepoints(text):
    return " ".join(f"U+{ord(ch):04X}" for ch in text)


# ──────────────────────────────────────────────────────────────────────────────
# 03.04b  ·  Quan sát mã Unicode — áp dụng
# ──────────────────────────────────────────────────────────────────────────────
# ord() đọc mã ký tự; normalize("NFD", ...) tách chữ và dấu để thấy các cách mã hóa khác nhau.
text_nfc = "Hòa"
text_nfd = unicodedata.normalize("NFD", text_nfc)
print("NFC:", text_nfc, "| length =", len(text_nfc))
print("NFD:", text_nfd, "| length =", len(text_nfd))
print("\nNFC codepoints:")
print(show_codepoints(text_nfc))
print("\nNFD codepoints:")
print(show_codepoints(text_nfd))
print("\nEquality:", text_nfc == text_nfd)

# ──────────────────────────────────────────────────────────────────────────────
# unicodedata.normalize("NFC", text) đưa chữ và dấu về dạng thống nhất, không xóa dấu tiếng
# Việt.
# Ví dụ: Chữ o + dấu huyền rời → ò.

# ──────────────────────────────────────────────────────────────────────────────
# 03.15  ·  Hàm chuẩn hóa cơ bản
# ──────────────────────────────────────────────────────────────────────────────
# normalize_basic() xử lý missing, chuẩn Unicode NFC và khoảng trắng; không tự xóa số hoặc dấu
# câu.
def normalize_basic(text):
    if text is None or pd.isna(text):
        return ""
    text = str(text)
    text = unicodedata.normalize("NFC", text)
    text = text.replace("\u00A0", " ")
    text = re.sub(r"\s+", " ", text)
    return text.strip()


# ──────────────────────────────────────────────────────────────────────────────
# 03.16  ·  Thử chuẩn hóa trên nhiều input
# ──────────────────────────────────────────────────────────────────────────────
# normalize_basic() được thử với chuỗi có khoảng trắng, None và chữ có dấu để kiểm tra phạm vi
# xử lý.
test_inputs = [
    "  Sinh viên\n đang   học NLP. ",
    None,
    "Hòa",
]
for x in test_inputs:
    print("INPUT :", repr(x))
    print("OUTPUT:", repr(normalize_basic(x)))
    print("-" * 50)


# ──────────────────────────────────────────────────────────────────────────────
# 03.17  ·  Hàm tách từ tiếng Việt
# ──────────────────────────────────────────────────────────────────────────────
# segment_vi() kết hợp chuẩn hóa và word_tokenize(format="text") để trả chuỗi token dễ vector
# hóa.
# Ví dụ: "sinh viên học" → "sinh_viên học" (minh họa).
def segment_vi(text):
    text = normalize_basic(text)
    if not text:
        return ""
    return word_tokenize(text, format="text")


# ──────────────────────────────────────────────────────────────────────────────

# nào.

# ──────────────────────────────────────────────────────────────────────────────
# 03.19a  ·  Bỏ dấu câu bằng Regex — hàm remove_punctuation()
# ──────────────────────────────────────────────────────────────────────────────
# Bỏ ký hiệu/dấu câu bằng Regex baseline để thử một cách tiền xử lý.
# Ví dụ: 'C++ và C#' → 'C và C'.
def remove_punctuation(text):
    text = re.sub(r"[^\w\s]", " ", text, flags=re.UNICODE)
    text = re.sub(r"\s+", " ", text).strip()
    return text


# ──────────────────────────────────────────────────────────────────────────────
# 03.19b  ·  Bỏ dấu câu bằng Regex — áp dụng
# ──────────────────────────────────────────────────────────────────────────────
# re.sub() thay các ký tự ngoài nhóm cho phép; cách này có thể làm mất tên hoặc ký hiệu có
# nghĩa.
# Ví dụ: "C++ và C#" → "C và C".
punctuation_test_cases = [
    "TP.HCM thu hút đầu tư.",

    "AI/ML đang thay đổi ngành công nghệ.",
]
for s in punctuation_test_cases:
    print("RAW :", s)
    print("DROP:", remove_punctuation(s))
    print("-" * 70)


# ──────────────────────────────────────────────────────────────────────────────
# 03.20a  ·  Bỏ số bằng Regex — hàm remove_numbers()
# ──────────────────────────────────────────────────────────────────────────────
# Bỏ các cụm số theo Regex baseline.
# Ví dụ: 'Có 12 người' → 'Có người'.
def remove_numbers(text):
    text = re.sub(r"\b\d[\d.,]*\b", " ", text)
    text = re.sub(r"\s+", " ", text).strip()
    return text


# ──────────────────────────────────────────────────────────────────────────────
# 03.20b  ·  Bỏ số bằng Regex — áp dụng
# ──────────────────────────────────────────────────────────────────────────────
# remove_numbers() xóa cụm số; dùng khi số không quan trọng với bài toán.
# Ví dụ: "Có 12 người" → "Có người".

# ──────────────────────────────────────────────────────────────────────────────
# 03.21  ·  Chọn danh sách stopword
# ──────────────────────────────────────────────────────────────────────────────
# Set chứa các token cần lọc; không mặc định bỏ từ phủ định nếu làm phân tích cảm xúc.
# Ví dụ: Bỏ "không" có thể làm đổi ý nghĩa câu.
STOPWORDS = {
    "và", "là", "của", "cho", "các", "một", "những",
    "được", "đang", "sẽ", "với", "ở", "từ", "trong",
    "trên", "theo", "đến", "về", "cũng", "đã", "khi",
    "thì", "này", "đó", "tại", "do", "vì",
}
print("Số stopwords:", len(STOPWORDS))


# ──────────────────────────────────────────────────────────────────────────────
# 03.22  ·  Hàm lọc stopword
# ──────────────────────────────────────────────────────────────────────────────
# remove_stopwords() loại các token có trong tập cho trước; lower() giúp so không phân biệt
# hoa/thường, join() ghép lại chuỗi.
def remove_stopwords(text, stopwords):
    tokens = str(text).split()
    normalized_stopwords = {str(word).lower() for word in stopwords}
    kept = [
        tok for tok in tokens
        if tok.lower() not in normalized_stopwords
    ]
    return " ".join(kept)


# ──────────────────────────────────────────────────────────────────────────────
# 03.23  ·  Cấu hình tiền xử lý
# ──────────────────────────────────────────────────────────────────────────────
# Các cờ True/False chọn có tách từ, lowercase hoặc bỏ số/dấu câu/stopword, giúp dùng lại cho
# bài khác.
DEFAULT_CONFIG = {
    "word_segment": True,
    "lowercase": True,
    "remove_punctuation": False,
    "remove_stopwords": False,
    "remove_numbers": False,
}


# ──────────────────────────────────────────────────────────────────────────────
# 03.24  ·  Ghép thành hàm tiền xử lý
# ──────────────────────────────────────────────────────────────────────────────
# preprocess_vi() kết hợp normalize_basic(), segment_vi() và các hàm lọc theo cấu hình được
# chọn.
def preprocess_vi(text, config=None):
    if config is None:
        config = DEFAULT_CONFIG.copy()
    else:
        cfg = DEFAULT_CONFIG.copy()
        cfg.update(config)
        config = cfg
    text = normalize_basic(text)
    if not text:
        return ""
    if config["word_segment"]:
        text = segment_vi(text)
    if config["lowercase"]:
        text = text.lower()
    if config["remove_numbers"]:
        text = remove_numbers(text)
    if config["remove_punctuation"]:
        text = remove_punctuation(text)
    if config["remove_stopwords"]:
        text = remove_stopwords(text, STOPWORDS)
    text = re.sub(r"\s+", " ", text).strip()
    return text

# ──────────────────────────────────────────────────────────────────────────────
# 03.31  ·  Đếm độ dài theo token
# ──────────────────────────────────────────────────────────────────────────────
# token_count() đếm các phần cách nhau bằng khoảng trắng; input rỗng trả 0.
# Ví dụ: "sinh_viên học bài" → 3 token.
def token_count(text):
    text = "" if text is None or pd.isna(text) else str(text).strip()
    if not text:
        return 0
    return len(text.split())


# ──────────────────────────────────────────────────────────────────────────────
# 03.32  ·  Thống kê độ dài

work_df["n_tokens_raw"] = work_df["text"].apply(token_count)
work_df["n_tokens_processed"] = work_df["processed_text"].apply(token_count)
display(
    work_df[

quality_summary = {
    "rows": len(work_df),
    "empty_processed": int(work_df["processed_text"].fillna("").str.strip().eq("").sum()),
    "processed_lt_3_tokens": int((work_df["n_tokens_processed"] < 3).sum()),
    "duplicate_raw_text": int(work_df["text"].duplicated().sum()),
    "duplicate_processed_text": int(work_df["processed_text"].duplicated().sum()),
}

# get_nonzero_features() dùng getrow() lấy vector một bài và trả các feature có số đếm lớn nhất.
def get_nonzero_features(
    X,
    feature_names,
    row_idx,
    top_n=20
):
    row = X.getrow(row_idx)
    pairs = [
        (feature_names[i], value)
        for i, value in zip(row.indices, row.data)
    ]
    pairs = sorted(
        pairs,
        key=lambda x: x[1],
        reverse=True
    )
    return pd.DataFrame(
        pairs[:top_n],
        columns=["feature", "value"]
    )

# 04.18  ·  Hàm tìm bài liên quan
# ──────────────────────────────────────────────────────────────────────────────
# find_similar_articles() dùng argsort() xếp điểm cosine và lấy các bài gần query nhất, bỏ chính
# query.
def find_similar_articles(
    doc_index,
    top_k,
    similarity_matrix,
    data
):
    scores = similarity_matrix[doc_index]
    ranked = np.argsort(scores)[::-1]
    ranked = [
        idx for idx in ranked
        if idx != doc_index
    ]
    ranked = ranked[:top_k]
    columns = ["title", "processed_text"]
    if "category" in data.columns:
        columns.insert(1, "category")
    result = data.iloc[ranked][columns].copy()
    result["similarity"] = [
        float(scores[idx])
        for idx in ranked
    ]
    return result

# ──────────────────────────────────────────────────────────────────────────────
# 04.24b  ·  Thử tác động của tiền xử lý — hàm matrix_density()
# ──────────────────────────────────────────────────────────────────────────────
# Đo tỷ lệ ô khác 0 trong ma trận để biết dữ liệu thưa đến mức nào.
# Ví dụ: Ma trận 10×100 có 50 ô khác 0 → density=50/1000=0.05, sparsity=0.95.
def matrix_density(matrix):
    return matrix.nnz / (matrix.shape[0] * matrix.shape[1])

# ──────────────────────────────────────────────────────────────────────────────
# 05.03b  ·  Tạo corpus mẫu — hàm build_sentences()
# ──────────────────────────────────────────────────────────────────────────────
# Tạo các câu mẫu bằng cách ghép chủ thể và ngữ cảnh để thử Word2Vec.
# Ví dụ: Chủ thể 'ngân_hàng' + ngữ cảnh 'hỗ_trợ doanh_nghiệp' → một câu mẫu.
def build_sentences(subjects, contexts, repeat=8):
    sentences = []
    for _ in range(repeat):
        random.shuffle(subjects)
        random.shuffle(contexts)
        for s in subjects:
            c = random.choice(contexts)
            sentences.append(f"{s} {c}")
    return sentences

# học từ cùng ngữ cảnh.

    build_sentences(finance_subjects, finance_contexts, repeat=10)
    + build_sentences(bank_subjects, bank_contexts, repeat=10)
    + build_sentences(tech_subjects, tech_contexts, repeat=10)
    + build_sentences(vehicle_subjects, vehicle_contexts, repeat=10)
    + build_sentences(education_subjects, education_contexts, repeat=10)
)

# ──────────────────────────────────────────────────────────────────────────────
# 05.22  ·  Hàm tạo vector cho cả bài
# ──────────────────────────────────────────────────────────────────────────────
# average_document_vector() lấy trung bình vector các token được biết; np.mean(axis=0) gộp theo
# các vector từ.
# Ví dụ: [1,3] và [3,5] → [2,4].
def average_document_vector(tokens, model):
    vectors = [
        model.wv[token]
        for token in tokens
        if token in model.wv
    ]
    if not vectors:
        return np.zeros(
            model.wv.vector_size,
            dtype=np.float32
        )
    return np.mean(vectors, axis=0)


# ──────────────────────────────────────────────────────────────────────────────
# 05.23  ·  Tạo ma trận vector bài
# ──────────────────────────────────────────────────────────────────────────────
# np.vstack() ghép vector từng bài thành ma trận số bài × số chiều, dùng cho tìm kiếm hoặc phân
# loại.
your_document_vectors = np.vstack([
    average_document_vector(text.split(), your_model)
    for text in your_df["processed_text"]
])
print("Document vector matrix shape:", your_document_vectors.shape)
assert your_document_vectors.shape == (len(your_df), your_model.wv.vector_size)


# ──────────────────────────────────────────────────────────────────────────────
# 05.24  ·  Hàm tìm bài bằng embedding
# ──────────────────────────────────────────────────────────────────────────────
# find_similar_documents() dùng cosine_similarity() so vector bài và argsort() xếp kết quả gần
# nhất.
def find_similar_documents(
    doc_index,
    document_vectors,
    data,
    top_k=5
):
    query = document_vectors[doc_index].reshape(1, -1)
    scores = cosine_similarity(query, document_vectors)[0]
    ranked = np.argsort(scores)[::-1]
    ranked = [
        idx for idx in ranked
        if idx != doc_index
    ][:top_k]
    columns = ["title"]
    if "category" in data.columns:
        columns.append("category")
    result = data.iloc[ranked][columns].copy()
    result["similarity"] = [
        float(scores[idx])
        for idx in ranked
    ]
    return result


# ──────────────────────────────────────────────────────────────────────────────
# 05.25  ·  Truy xuất các bài tương tự
# ──────────────────────────────────────────────────────────────────────────────

# là gợi ý liên quan.
if "category" in your_df.columns and your_df["category"].nunique() >= 3:
    query_indices = (
        your_df.reset_index()
        .groupby("category", sort=True)["index"]
        .first()
        .head(3)
        .tolist()
    )
else:
    query_indices = np.linspace(0, len(your_df) - 1, min(3, len(your_df)), dtype=int).tolist()
similar_document_results = {}
for idx in query_indices:
    result = find_similar_documents(idx, your_document_vectors, your_df, top_k=5)
    if "category" in your_df.columns:
        result["relevance_hint"] = np.where(
            result["category"].eq(your_df.iloc[idx]["category"]),
            "Relevant (same category)",
            "Review manually",
        )
    similar_document_results[idx] = result
    print("=" * 70)
    print(f"QUERY {idx}: {your_df.iloc[idx]['title']}")
    display(result)
print("Lưu ý: relevance_hint chỉ là gợi ý; cần đọc tiêu đề/nội dung để kết luận.")


# ──────────────────────────────────────────────────────────────────────────────
# 05.26  ·  Hàm đo độ bao phủ
# ──────────────────────────────────────────────────────────────────────────────
# document_coverage() tính tỷ lệ lượt token có vector trong Word2Vec.
# Ví dụ: Biết 8 trong 10 token → coverage=0.8.
def document_coverage(tokens, model):
    if not tokens:
        return 0.0
    in_vocab = sum(
        token in model.wv

def evaluate_model(name, model, X_train, y_train, X_test, y_test):

        ),
        "f1_macro": f1_score(
            y_test, pred, average="macro", zero_division=0
        ),
    }
    return row, pred, model

# 07.11b  ·  Hàm thử cấu hình — hàm run_experiment()
# ──────────────────────────────────────────────────────────────────────────────
# Thử một cấu hình TF-IDF/SVM và đánh giá trên cùng cách chia dữ liệu.
# Ví dụ: Có thể chỉ đổi unigram/bigram hoặc class_weight để kiểm tra tác động của từng lựa chọn.
def run_experiment(
    name,
    text_column=TEXT_COL,
    tfidf_kwargs=None,
    class_weight=None,
):
    if tfidf_kwargs is None:
        tfidf_kwargs = BASELINE_TFIDF.copy()
    vectorizer = TfidfVectorizer(**tfidf_kwargs)

    # fit_transform(): Học cách biến đổi từ dữ liệu rồi biến đổi chính dữ liệu đó.
    X_train = vectorizer.fit_transform(
        train_df[text_column].astype(str)
    )
    X_test = vectorizer.transform(
        test_df[text_column].astype(str)
    )
    model = LinearSVC(
        random_state=42,
        class_weight=class_weight,
    )

    # fit(): Cho đối tượng học các tham số từ dữ liệu: mô hình học quan hệ đặc trưng/nhãn,
    # vectorizer học vocabulary/IDF.
    model.fit(X_train, y_train)
    pred = model.predict(X_test)
    class_recalls = recall_score(
        y_test,
        pred,
        labels=labels,
        average=None,
        zero_division=0,
    )
    recall_by_class = dict(zip(labels, class_recalls))
    row = {
        "experiment": name,
        "text_input": text_column,
        "accuracy": accuracy_score(y_test, pred),
        "precision_macro": precision_score(
            y_test, pred, average="macro", zero_division=0
        ),
        "recall_macro": recall_score(
            y_test, pred, average="macro", zero_division=0
        ),
        "f1_macro": f1_score(
            y_test, pred, average="macro", zero_division=0
        ),
        "minority_recall": recall_by_class[MINORITY_CLASS],
        "n_features": X_train.shape[1],
    }
    return row, pred, vectorizer, model

# ──────────────────────────────────────────────────────────────────────────────
# 08.09a  ·  Xây CNN phân loại văn bản — hàm build_cnn_model()
# ──────────────────────────────────────────────────────────────────────────────
# Tạo CNN dùng Embedding và bộ lọc chuỗi để phân loại văn bản.
# Ví dụ: Nhận dãy token ID có cùng độ dài; cuối mạng trả một sigmoid score cho mỗi bài.
def build_cnn_model(
    vocab_size,
    max_len,
    embedding_dim=32,
    filters=64,
):
    model = keras.Sequential([
        layers.Input(shape=(max_len,)),
        layers.Embedding(
            input_dim=vocab_size,
            output_dim=embedding_dim,
        ),
        layers.Conv1D(
            filters=filters,
            kernel_size=3,
            activation="relu",
        ),
        layers.GlobalMaxPooling1D(),
        layers.Dropout(0.30),
        layers.Dense(32, activation="relu"),
        layers.Dropout(0.20),
        layers.Dense(1, activation="sigmoid"),
    ])
    model.compile(
        optimizer="adam",
        loss="binary_crossentropy",
        metrics=["accuracy"],
    )
    return model


# ──────────────────────────────────────────────────────────────────────────────
# 08.09b  ·  Xây CNN phân loại văn bản — áp dụng
# ──────────────────────────────────────────────────────────────────────────────
# Embedding() tạo vector token; Conv1D() tìm mẫu cục bộ; GlobalMaxPooling1D() lấy đặc trưng
# mạnh; Dense(sigmoid) dự đoán nhãn.

)

# ──────────────────────────────────────────────────────────────────────────────
# 08.13a  ·  Đánh giá CNN — hàm evaluate_dl_model()
# ──────────────────────────────────────────────────────────────────────────────
# Đổi sigmoid score thành nhãn theo ngưỡng 0.5 rồi đo accuracy/precision/recall/F1.
# Ví dụ: Score [0.2,0.8] → nhãn [0,1]; trả cả score để xem mức dự đoán.
def evaluate_dl_model(name, model, X_test, y_test):
    scores = model.predict(X_test, verbose=0).reshape(-1)
    pred = (scores >= 0.5).astype(int)
    row = {
        "model": name,
        "accuracy": accuracy_score(y_test, pred),
        "precision_macro": precision_score(
            y_test, pred, average="macro", zero_division=0
        ),
        "recall_macro": recall_score(
            y_test, pred, average="macro", zero_division=0
        ),
        "f1_macro": f1_score(
            y_test, pred, average="macro", zero_division=0
        ),
    }
    return row, pred, scores


# ──────────────────────────────────────────────────────────────────────────────

# ──────────────────────────────────────────────────────────────────────────────

# 08.15a  ·  Xây LSTM phân loại chuỗi — hàm build_lstm_model()
# ──────────────────────────────────────────────────────────────────────────────
# Tạo mạng LSTM để phân loại dựa trên chuỗi token theo thứ tự.
# Ví dụ: Dùng cùng input token ID như CNN, thay cách rút đặc trưng bằng LSTM.
def build_lstm_model(
    vocab_size,
    max_len,
    embedding_dim=32,
    lstm_units=32,
):
    model = keras.Sequential([
        layers.Input(shape=(max_len,)),
        layers.Embedding(
            input_dim=vocab_size,
            output_dim=embedding_dim,
            mask_zero=True,
        ),
        layers.LSTM(
            lstm_units,
            dropout=0.20,
        ),
        layers.Dense(32, activation="relu"),
        layers.Dropout(0.30),
        layers.Dense(1, activation="sigmoid"),
    ])
    model.compile(
        optimizer="adam",
        loss="binary_crossentropy",
        metrics=["accuracy"],
    )
    return model


# ──────────────────────────────────────────────────────────────────────────────
# 08.15b  ·  Xây LSTM phân loại chuỗi — áp dụng
# ──────────────────────────────────────────────────────────────────────────────
# Embedding() tạo vector token; LSTM() học thông tin theo thứ tự; Dense(sigmoid) dự đoán nhãn.
