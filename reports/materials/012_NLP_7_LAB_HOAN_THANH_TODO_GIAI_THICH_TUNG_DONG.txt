# 1) IMPORT — khai báo thư viện cung cấp các chức năng NLP cho tiếng Việt.
#    Cần cài 'underthesea' trước khi chạy: %pip install underthesea
# GIẢI THÍCH DÒNG DƯỚI: Nạp re: xử lý chuỗi bằng biểu thức chính quy (Regex).
import re
# GIẢI THÍCH DÒNG DƯỚI: Nạp pandas và đặt tên ngắn pd: đọc CSV, xử lý bảng DataFrame.
import pandas as pd
# GIẢI THÍCH DÒNG DƯỚI: Lấy display() để hiển thị bảng đẹp trong Notebook.
from IPython.display import display
# GIẢI THÍCH DÒNG DƯỚI: Nạp các hàm xử lý tiếng Việt (tách câu, tách từ, POS, NER).
from underthesea import sent_tokenize, word_tokenize, pos_tag, ner

# 2) DỮ LIỆU DEMO — đoạn dưới lấy nguyên trong Lab 01 của thầy.
# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào text; bước sau có thể dùng lại biến này.
text = """
Ngày 12/8/2026, Công ty ABC khai trương văn phòng mới tại Đà Nẵng.
Giám đốc Nguyễn Văn An cho biết công ty dự kiến tuyển thêm 100 kỹ sư phần mềm trong năm nay.
Doanh nghiệp sẽ hợp tác với Trường Đại học Công nghệ Thông tin và Truyền thông Việt - Hàn.
Mai công ty sẽ bắt đầu nhận hồ sơ.
Ứng viên có thể liên hệ qua email tuyendung@example.com hoặc số điện thoại 0905123456.
""".strip()
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print(text)


# 3) SENTENCE SEGMENTATION — cắt cả đoạn thành nhiều câu; len đếm số câu.
# GIẢI THÍCH DÒNG DƯỚI: Chia cả đoạn văn thành từng câu nhỏ.
sentences = sent_tokenize(text)
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("Số câu:", len(sentences))
# GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
for i, s in enumerate(sentences, 1):
    # GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
    print(f"{i}. {s}")

# 4) WORD SEGMENTATION — so sánh .split() với tách từ tiếng Việt của underthesea.
# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào sample_sentence; bước sau có thể dùng lại biến này.
sample_sentence = "Giám đốc Nguyễn Văn An cho biết công ty dự kiến tuyển thêm 100 kỹ sư phần mềm trong năm nay."
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("\nsplit():", sample_sentence.split())
# GIẢI THÍCH DÒNG DƯỚI: In/hiển thị kết quả của word_tokenize để kiểm tra dữ liệu hoặc đánh giá mô hình.
print("word_tokenize():", word_tokenize(sample_sentence))
# GIẢI THÍCH DÒNG DƯỚI: In/hiển thị kết quả của word_tokenize để kiểm tra dữ liệu hoặc đánh giá mô hình.
print("word_tokenize(format='text'):", word_tokenize(sample_sentence, format="text"))

# 5) POS TAGGING — trả danh sách (token, loại từ).
#    N: danh từ, Np: danh từ riêng, V: động từ, A: tính từ, M: số từ.
# GIẢI THÍCH DÒNG DƯỚI: Gán nhãn từ loại cho từng token: danh từ, động từ, tính từ...
pos_result = pos_tag(sample_sentence)
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("\nPOS:")
# GIẢI THÍCH DÒNG DƯỚI: Tạo bảng DataFrame để dễ xem và xử lý dữ liệu.
display(pd.DataFrame(pos_result, columns=["Token", "POS"]))

# 6) NER — tìm thực thể tên người (PER), địa điểm (LOC), tổ chức (ORG).
#    Thường trả (token, POS, Chunk, NER); có thể có 'O' nếu không phải thực thể.
# GIẢI THÍCH DÒNG DƯỚI: Nhận diện thực thể như người, địa điểm, tổ chức từ văn bản.
ner_result = ner(text)
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("\nNER (30 token đầu):")
# GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
for item in ner_result[:30]:
    # GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
    print(item)
# GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
if ner_result and len(ner_result[0]) == 4:
    # GIẢI THÍCH DÒNG DƯỚI: Tạo bảng DataFrame để dễ xem và xử lý dữ liệu.
    ner_df = pd.DataFrame(ner_result, columns=["Token", "POS", "Chunk", "NER"])
else:
    # GIẢI THÍCH DÒNG DƯỚI: Tạo bảng DataFrame để dễ xem và xử lý dữ liệu.
    ner_df = pd.DataFrame(ner_result)
# GIẢI THÍCH DÒNG DƯỚI: Hiển thị bảng hoặc đối tượng rõ ràng trong Notebook.
display(ner_df.head())

# 7) REGEX — không cần model AI: tìm email / số điện thoại theo mẫu cố định.
# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào email_pattern; bước sau có thể dùng lại biến này.
email_pattern = r"\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b"
# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào phone_pattern; bước sau có thể dùng lại biến này.
phone_pattern = r"\b0\d{9}\b"
# GIẢI THÍCH DÒNG DƯỚI: In/hiển thị kết quả của re.findall để kiểm tra dữ liệu hoặc đánh giá mô hình.
print("\nEmail:", re.findall(email_pattern, text))
# GIẢI THÍCH DÒNG DƯỚI: In/hiển thị kết quả của re.findall để kiểm tra dữ liệu hoặc đánh giá mô hình.
print("Điện thoại:", re.findall(phone_pattern, text))

# 8) CÂU KHÓ — kiểm tra cùng từ 'Mai' / 'Bình' có thể mang nghĩa khác nhau.
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho challenge_sentences; các phần tử có thể truy cập khi xử lý tiếp.
challenge_sentences = [
    "Mai tôi đi Đà Nẵng.",
    "Mai đang học NLP tại trường.",
    "Hôm nay Bình đến Hòa Bình.",
    "Công ty ABC tuyển dụng kỹ sư AI.",
]
# GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
for s in challenge_sentences:
    # GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
    print("\nCÂU:", s)
    # GIẢI THÍCH DÒNG DƯỚI: In/hiển thị kết quả của ner để kiểm tra dữ liệu hoặc đánh giá mô hình.
    print("NER:", ner(s))

student_text = """

# GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
for i, s in enumerate(student_sentences, 1):
    # GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
    print(i, s)

for s in student_sentences[:3]:

    print("TÁCH TỪ:", word_tokenize(s, format="text"))
# GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.

tricky_cases = [
    ("Mai tôi đi Đà Nẵng.", "NER", "Mai nói về ngày mai, không phải tên người"),
    ("Mai đang học NLP tại trường.", "NER", "Mai có thể là tên riêng của một người"),
    ("Công ty ABC tuyển dụng kỹ sư AI.", "NER", "ABC có thể là tên tổ chức"),
]
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho analysis_records; các phần tử có thể truy cập khi xử lý tiếp.

for sentence, task, expected in tricky_cases:
    # GIẢI THÍCH DÒNG DƯỚI: Nhận diện thực thể như người, địa điểm, tổ chức từ văn bản.

    # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.

        "Text / Token": sentence,

        "Model output": str(predicted),

    })
# GIẢI THÍCH DÒNG DƯỚI: Tạo bảng DataFrame để dễ xem và xử lý dữ liệu.
student_error_analysis = pd.DataFrame(analysis_records)
# GIẢI THÍCH DÒNG DƯỚI: Hiển thị bảng hoặc đối tượng rõ ràng trong Notebook.
display(student_error_analysis)

# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào SAMPLE_CATEGORY_HTML; bước sau có thể dùng lại biến này.
SAMPLE_CATEGORY_HTML = """

    </section>
</body>
</html>
"""

# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("Đã tạo HTML mẫu dùng offline.")


# 3) PARSE HTML: từ HTML string -> cấu trúc cây soup.
# GIẢI THÍCH DÒNG DƯỚI: Chuyển chuỗi HTML thành cây phần tử để tìm nội dung.
soup = BeautifulSoup(SAMPLE_ARTICLE_HTML, "html.parser")
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("Tiêu đề tab HTML:", soup.title.get_text(strip=True))
# GIẢI THÍCH DÒNG DƯỚI: Tìm phần tử HTML đầu tiên theo bộ chọn CSS.
category = soup.select_one(".category").get_text(" ", strip=True)
# GIẢI THÍCH DÒNG DƯỚI: Tìm phần tử HTML đầu tiên theo bộ chọn CSS.
title = soup.select_one(".title").get_text(" ", strip=True)
# GIẢI THÍCH DÒNG DƯỚI: Tìm phần tử HTML đầu tiên theo bộ chọn CSS.
description = soup.select_one(".description").get_text(" ", strip=True)
# GIẢI THÍCH DÒNG DƯỚI: Tìm phần tử HTML đầu tiên theo bộ chọn CSS.
published_at = soup.select_one(".published").get_text(" ", strip=True)
# GIẢI THÍCH DÒNG DƯỚI: Lấy danh sách tất cả phần tử HTML khớp bộ chọn CSS.
paragraphs = [p.get_text(" ", strip=True) for p in soup.select(".article-body p")]
# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào text; bước sau có thể dùng lại biến này.
text = " ".join(paragraphs)
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("TITLE:", title)
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("BODY:", text)

# 4) CLEANING — bỏ xuống dòng / nhiều khoảng trắng, CHƯA làm stopwords/segment.
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm clean_extracted_text(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def clean_extracted_text(value):
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if value is None:
        # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
        return ""
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return re.sub(r"\s+", " ", str(value)).strip()

# 5) RECORD — 1 bài báo = 1 dictionary, sau đó chuyển thành hàng của DataFrame.
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho demo_record; các phần tử có thể truy cập khi xử lý tiếp.
demo_record = {

    "source": "Demo News",
    "category": clean_extracted_text(category),
    "title": clean_extracted_text(title),
    "description": clean_extracted_text(description),
    "text": clean_extracted_text(text),
    "published_at": published_at,
    "crawled_at": datetime.now().isoformat(timespec="seconds"),
}
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("\nRECORD:", demo_record)

# 6) LỌC LINK BÀI VIẾT — bỏ '/about-us' và URL trùng, đổi sang absolute URL.
# GIẢI THÍCH DÒNG DƯỚI: Chuyển chuỗi HTML thành cây phần tử để tìm nội dung.
category_soup = BeautifulSoup(SAMPLE_CATEGORY_HTML, "html.parser")
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm is_article_url(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def is_article_url(url):
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if not url:
        # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
        return False
    # GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho allowed_prefixes; các phần tử có thể truy cập khi xử lý tiếp.
    allowed_prefixes = ("/kinh-doanh/", "/cong-nghe/", "/the-thao/")
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return url.endswith(".html") and url.startswith(allowed_prefixes)
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho article_links; các phần tử có thể truy cập khi xử lý tiếp.
article_links = []
# GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
for a in category_soup.select("a[href]"):
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào href; bước sau có thể dùng lại biến này.
    href = a.get("href")
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if is_article_url(href):
        # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
        article_links.append(href)
# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào article_links; bước sau có thể dùng lại biến này.
article_links = list(dict.fromkeys(article_links))
# GIẢI THÍCH DÒNG DƯỚI: Nối địa chỉ website gốc với liên kết tương đối để ra URL đầy đủ.
absolute_links = [urljoin("https://demo.local", u) for u in article_links]
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("\nARTICLE LINKS:", absolute_links)

# 7) SELECTOR -> trích 5 trường. Các selector này ĐÚNG CHO HTML DEMO, không mặc định đúng trên website khác.
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho SELECTORS; các phần tử có thể truy cập khi xử lý tiếp.
SELECTORS = {
    "category": ".category", "title": ".title", "description": ".description",
    "published_at": ".published", "paragraphs": ".article-body p",
}
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm extract_text_or_empty(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def extract_text_or_empty(soup, selector):
    # GIẢI THÍCH DÒNG DƯỚI: Tìm phần tử HTML đầu tiên theo bộ chọn CSS.
    node = soup.select_one(selector) if selector else None
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return clean_extracted_text(node.get_text(" ", strip=True)) if node else ""

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm parse_article_html(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def parse_article_html(html, url="", source="", selectors=None):
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào selectors; bước sau có thể dùng lại biến này.
    selectors = selectors or {}
    # GIẢI THÍCH DÒNG DƯỚI: Chuyển chuỗi HTML thành cây phần tử để tìm nội dung.
    document = BeautifulSoup(html, "html.parser")
    # GIẢI THÍCH DÒNG DƯỚI: Lấy danh sách tất cả phần tử HTML khớp bộ chọn CSS.
    paragraphs = [clean_extracted_text(p.get_text(" ", strip=True))
                  for p in document.select(selectors.get("paragraphs", ""))]
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return {
        "url": url, "source": source,
        "category": extract_text_or_empty(document, selectors.get("category", "")),
        "title": extract_text_or_empty(document, selectors.get("title", "")),
        "description": extract_text_or_empty(document, selectors.get("description", "")),
        "text": clean_extracted_text(" ".join(p for p in paragraphs if p)),
        "published_at": extract_text_or_empty(document, selectors.get("published_at", "")),
        "crawled_at": datetime.now().isoformat(timespec="seconds"),
    }

# 8) VALIDATE — file gốc dùng min_text_length=300 để lọc bài quá ngắn.
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm validate_article(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def validate_article(record, min_text_length=300):
    # GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho problems; các phần tử có thể truy cập khi xử lý tiếp.
    problems = []
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if not record.get("url"): problems.append("missing_url")
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if not record.get("title"): problems.append("missing_title")
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào body; bước sau có thể dùng lại biến này.
    body = record.get("text", "")
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if not body: problems.append("empty_text")
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    elif len(body) < min_text_length: problems.append("text_too_short")
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return {"is_valid": not problems, "problems": problems}

# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào record; bước sau có thể dùng lại biến này.
record = parse_article_html(SAMPLE_ARTICLE_HTML,
                            url=demo_record["url"], source="Demo News", selectors=SELECTORS)
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
print("\nVALIDATE 300:", validate_article(record, 300))

# 9) CRAWL WEBSITE THẬT — GIỮ ĐÚNG CƠ CHẾ safe_get/crawl_many TRONG BÀI GỐC.
#    KHÔNG GỌI trong bài offline vì example.com/demo.local không phải website thật để crawl.
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho DEFAULT_HEADERS; các phần tử có thể truy cập khi xử lý tiếp.
DEFAULT_HEADERS = {"User-Agent": "Mozilla/5.0 (compatible; NLP-Course-Lab/1.0)"}
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm safe_get(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def safe_get(url, timeout=15):
    # GIẢI THÍCH DÒNG DƯỚI: Gửi yêu cầu HTTP GET để tải nội dung website (cần Internet).
    response = requests.get(url, headers=DEFAULT_HEADERS, timeout=timeout)
    # GIẢI THÍCH DÒNG DƯỚI: Báo lỗi nếu phản hồi HTTP không thành công.
    response.raise_for_status()
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return response

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm crawl_article(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def crawl_article(url, source, selectors):
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào response; bước sau có thể dùng lại biến này.
    response = safe_get(url)
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return parse_article_html(response.text, url=url, source=source, selectors=selectors)

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm crawl_many(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def crawl_many(urls, source, selectors, delay_seconds=1.0, min_text_length=300):
    # GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho articles, errors; các phần tử có thể truy cập khi xử lý tiếp.
    articles, errors = [], []
    # GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
    for i, url in enumerate(urls, start=1):
        # GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.
        print(f"[{i}/{len(urls)}] {url}")
        # GIẢI THÍCH DÒNG DƯỚI: Thử chạy đoạn dễ phát sinh lỗi, chuyển sang except nếu có ngoại lệ.
        try:
            # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào rec; bước sau có thể dùng lại biến này.
            rec = crawl_article(url, source=source, selectors=selectors)
            # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào check; bước sau có thể dùng lại biến này.
            check = validate_article(rec, min_text_length=min_text_length)
            # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
            if check["is_valid"]: articles.append(rec)
            # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
            else: errors.append({"url": url, "error": ",".join(check["problems"])})
        except Exception as exc:
            # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
            errors.append({"url": url, "error": type(exc).__name__ + ": " + str(exc)})
        # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
        if delay_seconds > 0: time.sleep(delay_seconds)
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return articles, errors

# 10) DATA QUALITY — kiểm tra rỗng, trùng lặp, độ dài và phân bố nhãn.
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm clean_dataframe(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def clean_dataframe(data, min_text_length=300):
    # GIẢI THÍCH DÒNG DƯỚI: Tạo bản sao dữ liệu để thao tác mà không sửa bảng gốc.
    result = data.copy()
    # GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
    for col in ["category", "title", "description", "text"]:
        # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
        if col in result.columns:
            # GIẢI THÍCH DÒNG DƯỚI: Thay các giá trị bị thiếu bằng giá trị thay thế.
            result[col] = result[col].fillna("").map(clean_extracted_text)
    # GIẢI THÍCH DÒNG DƯỚI: Áp dụng thao tác xử lý chuỗi cho cả một cột văn bản.
    result = result[result["text"].str.len() >= min_text_length]
    # GIẢI THÍCH DÒNG DƯỚI: Loại bỏ dòng trùng nhau theo cột được chỉ định.
    result = result.drop_duplicates(subset=["url"], keep="first")
    # GIẢI THÍCH DÒNG DƯỚI: Loại bỏ dòng trùng nhau theo cột được chỉ định.
    result = result.drop_duplicates(subset=["text"], keep="first")
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return result.reset_index(drop=True)

# 3) CÁC HÀM BUILDING BLOCK — [HOÀN THIỆN TODO LAB03] theo gợi ý trong file gốc.
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm normalize_basic(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def normalize_basic(text):
    # None/NaN -> '' để hàm không bị lỗi, hỗ trợ unicode và whitespace.
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if text is None or pd.isna(text):
        # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
        return ""
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào text; bước sau có thể dùng lại biến này.
    text = unicodedata.normalize("NFC", str(text))
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào text; bước sau có thể dùng lại biến này.
    text = text.replace("\u00A0", " ")
    # GIẢI THÍCH DÒNG DƯỚI: Tìm mẫu bằng regex và thay bằng chuỗi khác (thường để làm sạch).
    text = re.sub(r"\s+", " ", text)
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return text.strip()

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm segment_vi(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def segment_vi(text):
    # underthesea gộp các âm tiết cùng từ bằng dấu gạch dưới, ví dụ 'sinh_viên'.
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào text; bước sau có thể dùng lại biến này.
    text = normalize_basic(text)
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if not text:
        # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
        return ""
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return word_tokenize(text, format="text")

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm remove_punctuation(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def remove_punctuation(text):
    # Baseline AGGRESSIVE trong Lab: xóa cả ký hiệu quan trọng như 'C++' hay '3,5%'.
    # GIẢI THÍCH DÒNG DƯỚI: Tìm mẫu bằng regex và thay bằng chuỗi khác (thường để làm sạch).
    text = re.sub(r"[^\w\s]", " ", text, flags=re.UNICODE)
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return re.sub(r"\s+", " ", text).strip()

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm remove_numbers(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def remove_numbers(text):
    # Baseline Lab: loại các cụm số; chưa xử lý hoàn hảo mọi ngày/tháng/thập phân.
    # GIẢI THÍCH DÒNG DƯỚI: Tìm mẫu bằng regex và thay bằng chuỗi khác (thường để làm sạch).
    text = re.sub(r"\b\d[\d.,]*\b", " ", text)
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return re.sub(r"\s+", " ", text).strip()

# Stopword danh sách minh họa theo demo Lab03. KHÔNG thêm 'không' (phủ định).
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho STOPWORDS; các phần tử có thể truy cập khi xử lý tiếp.
STOPWORDS = {"và", "là", "của", "cho", "các", "một", "những", "được", "đang", "sẽ", "với"}
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm remove_stopwords(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def remove_stopwords(text, stopwords):
    # GIẢI THÍCH DÒNG DƯỚI: Chia chuỗi thành các token dựa trên khoảng trắng.
    tokens = str(text).split()
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào kept; bước sau có thể dùng lại biến này.
    kept = [tok for tok in tokens if tok.lower() not in stopwords]
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return " ".join(kept)

# 4) CONFIG — cùng một pipeline, thay tham số để tạo 2 hướng xử lý.
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho DEFAULT_CONFIG; các phần tử có thể truy cập khi xử lý tiếp.
DEFAULT_CONFIG = {

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm preprocess_vi(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def preprocess_vi(text, config=None):
    # GIẢI THÍCH DÒNG DƯỚI: Tạo bản sao dữ liệu để thao tác mà không sửa bảng gốc.
    cfg = DEFAULT_CONFIG.copy()
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if config is not None:
        # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
        cfg.update(config)
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào text; bước sau có thể dùng lại biến này.
    text = normalize_basic(text)
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if not text:
        # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
        return ""
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if cfg["word_segment"]: text = segment_vi(text)
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if cfg["lowercase"]: text = text.lower()
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if cfg["remove_numbers"]: text = remove_numbers(text)
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if cfg["remove_punctuation"]: text = remove_punctuation(text)
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if cfg["remove_stopwords"]: text = remove_stopwords(text, STOPWORDS)
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return re.sub(r"\s+", " ", text).strip()

def get_nonzero_features(X, feature_names, row_idx, top_n=20):
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào row; bước sau có thể dùng lại biến này.
    row = X.getrow(row_idx)

    pairs = [(feature_names[i], val) for i, val in zip(row.indices, row.data)]
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return pd.DataFrame(sorted(pairs, key=lambda x: x[1], reverse=True)[:top_n],
                        columns=["feature", "value"])

def top_tfidf_features(X, names, row_idx, top_n=10):
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào row; bước sau có thể dùng lại biến này.
    row = X.getrow(row_idx)

    pairs = [(names[i], float(val)) for i, val in zip(row.indices, row.data)]
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return pd.DataFrame(sorted(pairs, key=lambda x: x[1], reverse=True)[:top_n],

def find_similar_articles(doc_index, top_k, similarity_matrix, data):
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào scores; bước sau có thể dùng lại biến này.
    scores = similarity_matrix[doc_index]
    # GIẢI THÍCH DÒNG DƯỚI: Tạo giá trị/biến theo kết quả của phép so sánh điều kiện.
    ranked = [idx for idx in np.argsort(scores)[::-1] if idx != doc_index][:top_k]
    # GIẢI THÍCH DÒNG DƯỚI: Tạo bản sao dữ liệu để thao tác mà không sửa bảng gốc.
    result = data.iloc[ranked][["title", "category", "processed_text"]].copy()
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào result["similarity"]; bước sau có thể dùng lại biến này.
    result["similarity"] = [float(scores[idx]) for idx in ranked]
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return result

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm build_sentences(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def build_sentences(subjects, contexts, repeat=8):
    # GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho sentences; các phần tử có thể truy cập khi xử lý tiếp.
    sentences = []

    # GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
    for _ in range(repeat):
        # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
        random.shuffle(subjects)
        # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
        random.shuffle(contexts)

        # GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
        for s in subjects:
            # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào c; bước sau có thể dùng lại biến này.
            c = random.choice(contexts)
            # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
            sentences.append(f"{s} {c}")

    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return sentences

# 9) DOCUMENT VECTOR: lấy trung bình vector các từ được nhận diện.
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm average_document_vector(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def average_document_vector(tokens, model):
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào vectors; bước sau có thể dùng lại biến này.
    vectors = [model.wv[token] for token in tokens if token in model.wv]
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if not vectors:
        # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
        return np.zeros(model.wv.vector_size, dtype=np.float32)
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return np.mean(vectors, axis=0)

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm document_coverage(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def document_coverage(tokens, model):
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if not tokens:
        # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
        return 0.0
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return sum(token in model.wv for token in tokens) / len(tokens)

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm find_similar_documents(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def find_similar_documents(doc_index, document_vectors, top_k=3):
    # GIẢI THÍCH DÒNG DƯỚI: Tính độ tương tự cosine giữa các vector.
    sims = cosine_similarity(document_vectors[doc_index:doc_index+1], document_vectors).ravel()
    # GIẢI THÍCH DÒNG DƯỚI: Tạo giá trị/biến theo kết quả của phép so sánh điều kiện.
    idxs = [idx for idx in np.argsort(sims)[::-1] if idx != doc_index][:top_k]
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return [(int(i), sample_texts[i], round(float(sims[i]), 4)) for i in idxs]
# GIẢI THÍCH DÒNG DƯỚI: In thông tin/kết quả hiện tại ra dưới cell.

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm evaluate_model(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def evaluate_model(name, model, X_train, y_train, X_test, y_test):
    # GIẢI THÍCH DÒNG DƯỚI: Huấn luyện hoặc học tham số dựa trên dữ liệu được cung cấp.
    model.fit(X_train, y_train)
    # GIẢI THÍCH DÒNG DƯỚI: Dùng model đã huấn luyện để dự đoán nhãn cho dữ liệu mới.
    pred = model.predict(X_test)
    # GIẢI THÍCH DÒNG DƯỚI: Tính tỷ lệ dự đoán đúng trên tổng số mẫu.
    row = {
        "model": name,
        "accuracy": accuracy_score(y_test, pred),
        "precision_macro": precision_score(y_test, pred, average="macro", zero_division=0),
        "recall_macro": recall_score(y_test, pred, average="macro", zero_division=0),
        "f1_macro": f1_score(y_test, pred, average="macro", zero_division=0),
    }
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return row, pred, model

results, predictions, models = [], {}, {}
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho classifiers; các phần tử có thể truy cập khi xử lý tiếp.
classifiers = {
    "MultinomialNB": MultinomialNB(),
    "LogisticRegression": LogisticRegression(max_iter=1000, random_state=42),
    "LinearSVC": LinearSVC(random_state=42),
}
# GIẢI THÍCH DÒNG DƯỚI: Vòng lặp: lấy từng phần tử từ danh sách hoặc dãy để xử lý lần lượt.
for name, classifier in classifiers.items():
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào row, pred, trained; bước sau có thể dùng lại biến này.
    row, pred, trained = evaluate_model(name, classifier, X_train, y_train, X_test, y_test)
    # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
    results.append(row)
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào predictions[name]; bước sau có thể dùng lại biến này.
    predictions[name] = pred
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào models[name]; bước sau có thể dùng lại biến này.
    models[name] = trained
# GIẢI THÍCH DÒNG DƯỚI: Tạo bảng DataFrame để dễ xem và xử lý dữ liệu.
results_df = pd.DataFrame(results).sort_values("f1_macro", ascending=False).reset_index(drop=True)

# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm run_experiment(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def run_experiment(name, text_column=TEXT_COL, tfidf_kwargs=None, class_weight=None):
    # GIẢI THÍCH DÒNG DƯỚI: Kiểm tra điều kiện; nếu đúng thì thực hiện phần code thụt vào bên dưới.
    if tfidf_kwargs is None:
        # GIẢI THÍCH DÒNG DƯỚI: Tạo bản sao dữ liệu để thao tác mà không sửa bảng gốc.
        tfidf_kwargs = BASELINE_TFIDF.copy()
    # GIẢI THÍCH DÒNG DƯỚI: Định nghĩa cách biểu diễn văn bản thành vector trọng số TF-IDF.
    vectorizer = TfidfVectorizer(**tfidf_kwargs)
    # GIẢI THÍCH DÒNG DƯỚI: Học bộ từ vựng từ dữ liệu train và đồng thời biến đổi thành số.
    X_train = vectorizer.fit_transform(train_df[text_column].astype(str))
    # GIẢI THÍCH DÒNG DƯỚI: Biến đổi dữ liệu bằng cách biểu diễn đã học, KHÔNG học lại.
    X_test = vectorizer.transform(test_df[text_column].astype(str))
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào model; bước sau có thể dùng lại biến này.
    model = LinearSVC(random_state=42, class_weight=class_weight)
    # GIẢI THÍCH DÒNG DƯỚI: Huấn luyện hoặc học tham số dựa trên dữ liệu được cung cấp.
    model.fit(X_train, y_train)
    # GIẢI THÍCH DÒNG DƯỚI: Dùng model đã huấn luyện để dự đoán nhãn cho dữ liệu mới.
    pred = model.predict(X_test)
    # GIẢI THÍCH DÒNG DƯỚI: Tính recall: trong số mẫu thật của một lớp, tìm đúng được bao nhiêu.
    class_recalls = recall_score(y_test, pred, labels=labels,
                                 average=None, zero_division=0)
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào recall_by_class; bước sau có thể dùng lại biến này.
    recall_by_class = dict(zip(labels, class_recalls))
    # GIẢI THÍCH DÒNG DƯỚI: Tính tỷ lệ dự đoán đúng trên tổng số mẫu.
    row = {
        "experiment": name, "text_input": text_column,
        "accuracy": accuracy_score(y_test, pred),
        "precision_macro": precision_score(y_test, pred, average="macro", zero_division=0),
        "recall_macro": recall_score(y_test, pred, average="macro", zero_division=0),
        "f1_macro": f1_score(y_test, pred, average="macro", zero_division=0),
        "f1_weighted": f1_score(y_test, pred, average="weighted", zero_division=0),
        "minority_recall": recall_by_class[MINORITY_CLASS],
        "n_features": X_train.shape[1],
    }
    # GIẢI THÍCH DÒNG DƯỚI: Trả kết quả cuối của hàm cho nơi đã gọi hàm này.
    return row, pred, vectorizer, model

# 7) BEFORE + 2 AFTER. TODO trong Lab07 đã được hoàn thiện bằng cấu hình đúng đề thầy.
# GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho experiments, experiment_predictions, experiment_objects; các phần tử có thể truy cập khi xử lý tiếp.
experiments, experiment_predictions, experiment_objects = [], {}, {}
# GIẢI THÍCH DÒNG DƯỚI: Định nghĩa hàm add_exp(...): gom thao tác thành một khối có thể gọi lại; thân hàm thụt vào.
def add_exp(name, **kwargs):
    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào row, pred, vec, model; bước sau có thể dùng lại biến này.
    row, pred, vec, model = run_experiment(name, **kwargs)
    # GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.

    # GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào experiment_predictions[name]; bước sau có thể dùng lại biến này.
    experiment_predictions[name] = pred
    # GIẢI THÍCH DÒNG DƯỚI: Khởi tạo dữ liệu/cấu hình cho experiment_objects[name]; các phần tử có thể truy cập khi xử lý tiếp.
    experiment_objects[name] = (vec, model)

# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào baseline_name; bước sau có thể dùng lại biến này.
baseline_name = "Baseline: (1,2), class_weight=None"
# GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.

# Exp 1 thay ngram_range từ (1,2) sang (1,1), KHÔNG đổi model.
# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào unigram_tfidf; bước sau có thể dùng lại biến này.

# GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
add_exp("Exp 1: unigram only", tfidf_kwargs=unigram_tfidf)
# Exp 2 chỉ đổi class_weight = balanced, giữ nguyên ngram (1,2).
# GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
add_exp('Exp 2: class_weight="balanced"', class_weight="balanced")
# ĐÃ HOÀN THIỆN BÀI TẬP 6 (Experiment 3): thử min_df=1, chỉ thay min_df so với baseline.
# GIẢI THÍCH DÒNG DƯỚI: Gán kết quả biểu thức vào min_df_one; bước sau có thể dùng lại biến này.
min_df_one = dict(BASELINE_TFIDF, min_df=1)
# GIẢI THÍCH DÒNG DƯỚI: Gọi hàm hoặc thực hiện thao tác và quan sát hiệu ứng/kết quả.
