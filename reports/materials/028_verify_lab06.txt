import json
import os
import sys
from pathlib import Path

import nbformat
from nbclient import NotebookClient

root = Path(__file__).resolve().parent
runtime = root / '.lab06_runtime'
kernel_dir = runtime / 'kernels' / 'lab06'
kernel_dir.mkdir(parents=True, exist_ok=True)
(kernel_dir / 'kernel.json').write_text(json.dumps({
    'argv': [sys.executable, '-m', 'ipykernel_launcher', '-f', '{connection_file}'],
    'display_name': 'Lab06 verification', 'language': 'python',
}), encoding='utf-8')
os.environ['JUPYTER_PATH'] = str(runtime)
os.environ['JUPYTER_RUNTIME_DIR'] = str(runtime / 'jupyter')
os.environ['IPYTHONDIR'] = str(runtime / 'ipython')
os.environ['MPLCONFIGDIR'] = str(runtime / 'matplotlib')
path = root / 'Lab06_Student_Guided_Text_Classification.ipynb'
nb = nbformat.read(path, as_version=4)
client = NotebookClient(nb, timeout=180, kernel_name='lab06', resources={'metadata': {'path': str(root)}})
client.execute()
nb.metadata.kernelspec = {'display_name': 'Python 3', 'language': 'python', 'name': 'python3'}
nbformat.validate(nb)
nbformat.write(nb, path)
codes = [cell for cell in nb.cells if cell.cell_type == 'code']
assert all(cell.execution_count is not None for cell in codes)
assert not any(out.output_type == 'error' for cell in codes for out in cell.outputs)
images = sum('image/png' in out.get('data', {}) for cell in codes for out in cell.outputs)
assert images >= 1, 'Missing confusion matrix image'
print(f'PASS: {len(codes)} code cells, {images} embedded PNG outputs; no execution errors.')
print((root / 'lab06_artifacts' / 'classification_results.csv').read_text(encoding='utf-8-sig'))
