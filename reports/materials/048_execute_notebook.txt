# -*- coding: utf-8 -*-
"""Execute notebook cells directly in python and save outputs into notebook json."""
import os
import sys
import io
import json
import base64
import ast
from pathlib import Path

os.chdir(r'd:\NLP\Lab09')

nb_path = r'd:\NLP\Lab09\Lab09_Student_Conversational_NLP_I.ipynb'

with open(nb_path, 'r', encoding='utf-8') as f:
    nb = json.load(f)

import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import pandas as pd
import numpy as np

# Intercept display
current_outputs = []

def custom_display(*args, **kwargs):
    for arg in args:
        if isinstance(arg, pd.DataFrame):
            current_outputs.append({
                "data": {
                    "text/html": [arg.to_html()],
                    "text/plain": [arg.to_string()]
                },
                "metadata": {},
                "output_type": "display_data"
            })
        elif hasattr(arg, '_repr_html_'):
            current_outputs.append({
                "data": {
                    "text/html": [arg._repr_html_()],
                    "text/plain": [str(arg)]
                },
                "metadata": {},
                "output_type": "display_data"
            })
        else:
            current_outputs.append({
                "data": {
                    "text/plain": [repr(arg)]
                },
                "metadata": {},
                "output_type": "display_data"
            })

# Global environment
gl = {
    'display': custom_display,
    '__name__': '__main__',
}

# Hook plt.show
original_show = plt.show

def custom_show(*args, **kwargs):
    fig = plt.gcf()
    buf = io.BytesIO()
    fig.savefig(buf, format='png', bbox_inches='tight', dpi=150)
    buf.seek(0)
    b64 = base64.b64encode(buf.read()).decode('utf-8')
    plt.close(fig)
    current_outputs.append({
        "data": {
            "image/png": b64,
            "text/plain": ["<Figure size ...>"]
        },
        "metadata": {
            "needs_background": "light"
        },
        "output_type": "display_data"
    })

plt.show = custom_show
gl['plt'] = plt

exec_count = 1

for cell_idx, cell in enumerate(nb['cells']):
    if cell['cell_type'] != 'code':
        continue
    
    code = ''.join(cell['source'])
    current_outputs = []
    
    # Capture stdout
    stdout_buf = io.StringIO()
    old_stdout = sys.stdout
    sys.stdout = stdout_buf
    
    try:
        # Check if last node is an expression
        tree = ast.parse(code)
        last_expr = None
        if tree.body and isinstance(tree.body[-1], ast.Expr):
            last_expr = tree.body.pop()
        
        # Exec statements
        if tree.body:
            compiled = compile(tree, filename='<cell>', mode='exec')
            exec(compiled, gl)
        
        # Eval last expression if any
        if last_expr is not None:
            compiled_expr = compile(ast.Expression(last_expr.value), filename='<cell>', mode='eval')
            res = eval(compiled_expr, gl)
            if res is not None:
                # If expression returns something, add as execute_result
                if isinstance(res, pd.DataFrame):
                    current_outputs.append({
                        "data": {
                            "text/html": [res.to_html()],
                            "text/plain": [res.to_string()]
                        },
                        "execution_count": exec_count,
                        "metadata": {},
                        "output_type": "execute_result"
                    })
                elif hasattr(res, '_repr_html_'):
                    current_outputs.append({
                        "data": {
                            "text/html": [res._repr_html_()],
                            "text/plain": [str(res)]
                        },
                        "execution_count": exec_count,
                        "metadata": {},
                        "output_type": "execute_result"
                    })
                else:
                    import pprint
                    formatted = pprint.pformat(res, sort_dicts=False, width=80)
                    current_outputs.append({
                        "data": {
                            "text/plain": [formatted]
                        },
                        "execution_count": exec_count,
                        "metadata": {},
                        "output_type": "execute_result"
                    })
    except Exception as e:
        import traceback
        tb = traceback.format_exc()
        current_outputs.append({
            "ename": type(e).__name__,
            "evalue": str(e),
            "output_type": "error",
            "traceback": tb.splitlines()
        })
        print(f"Error in cell {cell_idx}: {e}")
    finally:
        sys.stdout = old_stdout
    
    stdout_val = stdout_buf.getvalue()
    if stdout_val:
        current_outputs.insert(0, {
            "name": "stdout",
            "output_type": "stream",
            "text": stdout_val.splitlines(keepends=True)
        })
    
    cell['execution_count'] = exec_count
    cell['outputs'] = current_outputs
    print(f"Executed Code Cell {cell_idx} (In [{exec_count}]): {len(current_outputs)} output(s)")
    exec_count += 1

# Save notebook with outputs
with open(nb_path, 'w', encoding='utf-8') as f:
    json.dump(nb, f, ensure_ascii=False, indent=1)

print("\nNotebook executed and saved successfully with all outputs!")
