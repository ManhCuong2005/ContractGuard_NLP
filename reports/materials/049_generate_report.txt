def set_font(run, size=13, bold=False, italic=False):
    run.font.name = 'Times New Roman'
    run.font.size = Pt(size)
    run.font.color.rgb = RGBColor(0, 0, 0)
    run.bold = bold
    run.italic = italic
    run.element.rPr.rFonts.set(qn('w:eastAsia'), 'Times New Roman')

def add_heading_custom(text, level=1):
    p = doc.add_paragraph()
    if level == 1:
        p.alignment = WD_ALIGN_PARAGRAPH.CENTER
        run = p.add_run(text)
        set_font(run, size=16, bold=True)
    elif level == 2:
        run = p.add_run(text)
        set_font(run, size=14, bold=True)
    else:
        run = p.add_run(text)
        set_font(run, size=13, bold=True)
    return p

def add_text(text, bold=False, italic=False, size=13):
    p = doc.add_paragraph()
    run = p.add_run(text)
    set_font(run, size=size, bold=bold, italic=italic)
    return p

def add_code_block(code_text):
    """Add code as a formatted paragraph with monospace font and gray background."""
    p = doc.add_paragraph()
    # Set paragraph shading to light gray
    pPr = p._element.get_or_add_pPr()
    shading = pPr.makeelement(qn('w:shd'), {
        qn('w:fill'): 'F2F2F2',
        qn('w:val'): 'clear',
    })
    pPr.append(shading)
    run = p.add_run(code_text)
    run.font.name = 'Consolas'
    run.font.size = Pt(10)
    run.font.color.rgb = RGBColor(0, 0, 0)
    return p

code_task2 = """def extract_slots(text: str) -> dict[str, str]:
    result: dict[str, str] = {}
    raw = unicodedata.normalize('NFC', str(text)).strip()
    t = normalize(raw)
    # 2A: tìm vehicle
    if re.search(r'xe\\s+máy|xe\\s+ôm', t):
        result['vehicle'] = 'xe máy'
    elif re.search(r'ô\\s*tô|taxi|xe\\s+hơi|xe\\s+4\\s+chỗ', t):
        result['vehicle'] = 'ô tô'
    # 2B: tìm pickup/dropoff
    m = re.search(r'từ\\s+(.+?)\\s+(?:đến|tới)\\s+(.+)', raw, flags=re.I)
    if m:
        result['pickup'] = clean_place(m.group(1))
        result['dropoff'] = clean_place(m.group(2))
    return result"""

code_task4 = """def _handle_confirmation(self, text):
    t = normalize(text)
    yes = {'đồng ý', 'vâng', 'ừ', 'ok', 'xác nhận', 'chốt', 'được', 'có', 'đúng rồi'}
    no = {'không', 'không đồng ý', 'chưa', 'sai rồi', 'thôi'}
    if t in yes:
        summary = (f"Đã ghi nhận yêu cầu MÔ PHỎNG: {self.slots['vehicle']} "
                   f"từ {self.slots['pickup']} đến {self.slots['dropoff']}. "
                   "Không có chuyến xe thật nào được đặt.")
        self.reset()
        return summary
    if t in no:
        self.reset()
        return 'Đã hủy yêu cầu mô phỏng.'
    return 'Bạn vui lòng trả lời "đồng ý" để xác nhận hoặc "hủy" để bỏ yêu cầu nhé.'"""
