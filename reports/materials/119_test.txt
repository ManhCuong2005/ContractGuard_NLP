from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.pipeline import Pipeline

# Du lieu mau
texts = [
    "gia vang tang",
    "co phieu tang",
    "ngan hang giam lai suat",
    "thi truong chung khoan",
    "lap trinh python",
    "tri tue nhan tao",
    "cong nghe phan mem",
    "may tinh va robot"
]

labels = [
    "Kinh doanh",
    "Kinh doanh",
    "Kinh doanh",
    "Kinh doanh",
    "Cong nghe",
    "Cong nghe",
    "Cong nghe",
    "Cong nghe"
]

# Tao pipeline
model = Pipeline([
    ("tfidf", TfidfVectorizer()),
    ("classifier", MultinomialNB())
])

# Huan luyen
model.fit(texts, labels)

# Du doan
samples = [
    "gia vang va co phieu",
    "lap trinh phan mem"
]

predictions = model.predict(samples)

for text, label in zip(samples, predictions):
    print(text, "=>", label)

print("\nHOAN THANH TEST 3!")
