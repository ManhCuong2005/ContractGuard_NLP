# Pack figures without shrinking their text. Heights use the browser's 96 dpi.
used = 0
budget = 960
page_count = 1
def reserve(height):
    global used, page_count
    if used + height > budget and used > 0:
        previous = doc.paragraphs[-1] if doc.paragraphs else None
        if previous is not None and previous.style.name in ['Title','Heading 1','Heading 2']:
            previous.paragraph_format.page_break_before = True
            used = estimated_lines(previous.text)*20+15
        else:
            doc.add_page_break()
            used = 0
        page_count += 1
    used += height

font = ImageFont.truetype(r'C:\Windows\Fonts\times.ttf', 18)
def estimated_lines(text):
    lines=1; line=''
    for word in text.split():
        if font.getlength(line+' '+word) > 642:
            lines += 1; line = word
        else: line += ' '+word
    return lines
def para(text, bold=False, style=None):
    height = estimated_lines(text)*20 + (15 if style else 7)
    reserve(height)
    p = doc.add_paragraph(style=style)
    r = p.add_run(text); r.bold = bold or style is not None
    p.paragraph_format.widow_control = True
    if bold:
        p.paragraph_format.keep_with_next = True
    return p
def heading(text):
    reserve(60)
    global used
    used -= 60
    return para(text, style='Heading 1')
def pic(id):
    img = image_by_id[id]
    reserve(img['height']+9)
    p = doc.add_paragraph()

    p.paragraph_format.line_spacing = 1
    p.add_run().add_picture(str(WORK/'screenshots'/(id+'.png')), width=Cm(17))
    drawing=p._p.xpath('.//wp:docPr')
    if drawing: drawing[0].set('descr',img['label'])
