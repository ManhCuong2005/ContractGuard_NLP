from pathlib import Path
import importlib
import joblib
import numpy as np
import pandas as pd
import matplotlib

import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split, StratifiedKFold, cross_val_score
from sklearn.pipeline import Pipeline
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import classification_report, ConfusionMatrixDisplay, accuracy_score, f1_score

RANDOM_STATE = 42
DATA_FILE = Path('intents_vi.csv')
OUTPUT_DIR = Path(r'd:\NLP\Lab09\report_outputs')
OUTPUT_DIR.mkdir(exist_ok=True)

outputs = {}

# ==================== CELL B: Dataset ====================
print("=== Cell B: Dataset ===")
df = pd.read_csv(DATA_FILE)
out_b = []
out_b.append(f"Dataset: {df.shape} | Classes: {df['intent'].nunique()}")
out_b.append("")
out_b.append(df['intent'].value_counts().rename('n').to_frame().to_string())
out_b.append("")
out_b.append(df.sample(6, random_state=RANDOM_STATE).to_string())

X_train, X_test, y_train, y_test = train_test_split(
    df['text'], df['intent'], test_size=0.25, random_state=RANDOM_STATE,
    stratify=df['intent'],
)
out_b.append(f"\ntrain: {len(X_train)} | test: {len(X_test)}")
outputs['cell_B'] = '\n'.join(out_b)
print(outputs['cell_B'])

# ==================== CELL C: Intent Classifier ====================
print("\n=== Cell C: Intent Classifier ===")
intent_model = Pipeline([
    ('tfidf', TfidfVectorizer(analyzer='char_wb', ngram_range=(2, 5), lowercase=True)),
    ('clf', LogisticRegression(max_iter=2000, class_weight='balanced', random_state=RANDOM_STATE)),
])
intent_model.fit(X_train, y_train)
y_pred = intent_model.predict(X_test)
report_str = classification_report(y_test, y_pred, zero_division=0)
macro_f1 = round(f1_score(y_test, y_pred, average='macro'), 3)
joblib.dump(intent_model, 'intent_model.joblib')

out_c = []
out_c.append(report_str)
out_c.append(f"Test Macro F1: {macro_f1}")
out_c.append("Saved: intent_model.joblib")
outputs['cell_C'] = '\n'.join(out_c)
print(outputs['cell_C'])

# ==================== CELL C2: Confusion Matrix ====================
print("\n=== Cell C2: Confusion Matrix ===")
fig, ax = plt.subplots(figsize=(8, 6))
ConfusionMatrixDisplay.from_predictions(y_test, y_pred, xticks_rotation=50, cmap='Blues', ax=ax, colorbar=False)
plt.tight_layout()
plt.savefig(str(OUTPUT_DIR / 'confusion_matrix.png'), dpi=150, bbox_inches='tight')
plt.close()

error_df = errors.loc[errors.actual != errors.predicted].head(10)

print("\n=== Cell D: Predict Samples ===")
def predict_samples(model, samples):
    proba = model.predict_proba(samples)
    winners = np.argmax(proba, axis=1)
    return pd.DataFrame({
        'text': samples,
        'intent': [model.classes_[i] for i in winners],
        'max_probability': np.round(proba[np.arange(len(samples)), winners], 3),
    })

samples = [
    'Đặt xe giúp tôi',
    'Cho tôi xe máy từ Cầu Rồng tới Mỹ Khê',
    'Tôi cần hướng dẫn',
    'Hôm nay có mưa không',
    'Cầu Rồng',
]
pred_df = predict_samples(intent_model, samples)
outputs['cell_D'] = pred_df.to_string()

# ==================== CELL H: Mini Experiment ====================
print("\n=== Cell H: Mini Experiment ===")
cv = StratifiedKFold(n_splits=3, shuffle=True, random_state=RANDOM_STATE)
trial_results = []
for ngrams in [(2, 5), (2, 4)]:
    trial = Pipeline([
        ('tfidf', TfidfVectorizer(analyzer='char_wb', ngram_range=ngrams)),
        ('clf', LogisticRegression(max_iter=2000, class_weight='balanced', random_state=RANDOM_STATE)),
    ])
    scores = cross_val_score(trial, X_train, y_train, cv=cv, scoring='f1_macro')
    trial_results.append({'ngram_range':str(ngrams), 'mean_cv_macro_f1':round(float(scores.mean()), 3),
                          'std':round(float(scores.std()), 3)})
trial_df = pd.DataFrame(trial_results)
